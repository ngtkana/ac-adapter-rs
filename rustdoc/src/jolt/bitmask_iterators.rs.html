4/// ちょうど $k$ ビット立っている $n$ ビットのビットマスクを昇順に列挙する。
5///
6/// $[0, 2^n)$ の整数のうち popcount が $k$ であるもの全体を昇順に返す。
7/// 立っている最下位ビットの塊を 1 段上にずらす古典的な手法（Gosper's hack）で、
8/// 各要素を $O(1)$ で次に進める。
9///
10/// # 計算量
11///
12/// 全体で $O(\binom{n}{k})$。
13///
14/// # 例
15///
16/// ```
17/// use jolt::bitmask_combinations;
18///
19/// assert_eq!(bitmask_combinations::<u32>(3, 2).collect::<Vec<_>>(), vec![
20///     3, 5, 6
21/// ]);
22/// ```
23pub fn bitmask_combinations<T: Unsigned>(n: u32, k: u32) -> BitmaskCombinations<T> {
24    assert!(k < T::bit_length() && n <= T::bit_length());
25    BitmaskCombinations {
26        n,
27        bs: (T::ONE << k) - T::ONE,
28        finished: false,
29    }
30}
31
32/// [`bitmask_combinations`] が返すイテレータ。
33#[derive(Clone, Debug, Default, Hash, PartialEq, Eq)]
34pub struct BitmaskCombinations<T> {
35    n: u32,
36    bs: T,
37    finished: bool,
38}
39impl<T: Unsigned> Iterator for BitmaskCombinations<T> {
40    type Item = T;
41
42    fn next(&mut self) -> Option<Self::Item> {
43        if self.finished {
44            return None;
45        }
46        let res = Some(self.bs);
47        if self.bs == T::ZERO {
48            self.finished = true;
49        } else {
50            let x = self.bs & self.bs.wrapping_neg();
51            if self.bs > T::MAX - x {
52                // `self.bs + x` would overflow `T`: `self.bs` was the last combination.
53                self.finished = true;
54            } else {
55                let y = self.bs + x;
56                let next_bs = (((self.bs & !y) / x) >> 1) | y;
57                if next_bs > i2powm1::<T>(self.n) {
58                    self.finished = true;
59                } else {
60                    self.bs = next_bs;
61                }
62            }
63        }
64        res
65    }
66}

68/// `bs` の部分ビットマスク（`bs` 自身と $0$ を含む）を昇順に列挙する。
69///
70/// $s \mathbin{\&} \mathrm{bs} = s$ を満たす整数 $s$ 全体を昇順に返す。
71/// 部分マスクを $1$ ずつ減らして `bs` との AND を取っていく古典的な手法を、
72/// `bs` との XOR で反転させることで昇順化している。各要素 $O(1)$。
73///
74/// # 計算量
75///
76/// 全体で $O(2^{\mathrm{popcount}(\mathrm{bs})})$。
77///
78/// # 例
79///
80/// ```
81/// use jolt::bitmask_subsets;
82///
83/// assert_eq!(bitmask_subsets(10u32).collect::<Vec<_>>(), vec![
84///     0, 2, 8, 10
85/// ]);
86/// ```
87pub fn bitmask_subsets<T: Unsigned>(bs: T) -> BitmaskSubsets<T> {
88    BitmaskSubsets {
89        bs,
90        full: bs,
91        finished: false,
92    }
93}
94
95/// [`bitmask_subsets`] が返すイテレータ。
96#[derive(Clone, Debug, Default, Hash, PartialEq, Eq)]
97pub struct BitmaskSubsets<T> {
98    bs: T,
99    full: T,
100    finished: bool,
101}
102impl<T: Unsigned> Iterator for BitmaskSubsets<T> {
103    type Item = T;
104
105    fn next(&mut self) -> Option<Self::Item> {
106        if self.finished {
107            return None;
108        }
109        let res = Some(self.bs ^ self.full);
110        if self.bs == T::ZERO {
111            self.finished = true;
112        } else {
113            self.bs -= T::ONE;
114            self.bs &= self.full;
115        }
116        res
117    }
118}
