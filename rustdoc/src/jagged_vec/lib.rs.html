1//! インデックス依存の長さを持つ、skew な多次元 `Vec` を生成するマクロです。
2//!
3//! # Examples
4//!
5//! ```
6//! use jagged_vec::jagged_vec;
7//!
8//! let n = 3;
9//! let v = jagged_vec![n; |i| n - i, |i, j| n - i - j; 0];
10//! assert_eq!(v, vec![
11//!     vec![vec![0, 0, 0], vec![0, 0], vec![0]],
12//!     vec![vec![0, 0], vec![0]],
13//!     vec![vec![0]],
14//! ]);
15//! ```
16
17/// skew な多次元 `Vec` を生成します。
18///
19/// # 構文
20///
21/// `jagged_vec![len0; f1, f2, ..., fk; init]`
22///
23/// - `len0`: 最外次元（0 次元目）の長さ
24/// - `f1, ..., fk`: 各次元の長さを決めるクロージャ。`fj` は外側 `j` 個のインデックス
25///   `(i0, ..., i_{j-1})` を引数に取り、`j` 次元目の長さを `usize` で返す
26/// - `init`: 末端要素の初期値（`Clone`）。1 度だけ評価され、各要素に `clone` されます
27///
28/// クロージャを省略すると、通常の `vec![init; len0]` と同じ 1 次元 `Vec` になります。
29///
30/// 依存関係が型では表現できない可変個・可変引数のクロージャ列を扱うため、
31/// ネストしたループの組み立てはマクロが担い、各次元の長さ計算・要素の複製は
32/// 通常の Rust の型（クロージャ・`Clone`）に委ねています。
33///
34/// # Examples
35///
36/// 1 次元（通常の `vec!` と同じ）:
37///
38/// ```
39/// use jagged_vec::jagged_vec;
40///
41/// let v = jagged_vec![3; 0];
42/// assert_eq!(v, vec![0, 0, 0]);
43/// ```
44///
45/// 2 次元（三角形）:
46///
47/// ```
48/// use jagged_vec::jagged_vec;
49///
50/// let v = jagged_vec![3; |i| i + 1; 0];
51/// assert_eq!(v, vec![vec![0], vec![0, 0], vec![0, 0, 0]]);
52/// ```
53///
54/// 3 次元（`i + j + k < n` を満たす形の skew な `Vec`）:
55///
56/// ```
57/// use jagged_vec::jagged_vec;
58///
59/// let n = 3;
60/// let v = jagged_vec![n; |i| n - i, |i, j| n - i - j; 0];
61/// assert_eq!(v, vec![
62///     vec![vec![0, 0, 0], vec![0, 0], vec![0]],
63///     vec![vec![0, 0], vec![0]],
64///     vec![vec![0]],
65/// ]);
66/// ```
67#[macro_export]
68macro_rules! jagged_vec {
69    ($len0:expr; $init:expr) => {
70        ::std::vec![$init; $len0]
71    };
72    ($len0:expr; $($f:expr),+ $(,)?; $init:expr) => {{
73        let __jagged_vec_init = $init;
74        $crate::__jagged_vec_body!(
75            &__jagged_vec_init;
76            $len0;
77            [];
78            [__jv0, __jv1, __jv2, __jv3, __jv4, __jv5, __jv6, __jv7];
79            $($f),+
80        )
81    }};
82}
83
84#[doc(hidden)]
85#[macro_export]
86macro_rules! __jagged_vec_body {
87    ($init:expr; $len:expr; [$($acc:ident),*]; [$($pool:ident),*]; ) => {{
88        let __jv_len: usize = $len;
89        ::std::vec![::std::clone::Clone::clone($init); __jv_len]
90    }};
91    ($init:expr; $len:expr; [$($acc:ident),*]; [$pool_head:ident $(, $pool_tail:ident)*]; $head:expr $(, $tail:expr)*) => {{
92        let __jv_len: usize = $len;
93        (0..__jv_len)
94            .map(|$pool_head| {
95                let __jv_next_len: usize = ($head)($($acc,)* $pool_head);
96                $crate::__jagged_vec_body!(
97                    $init;
98                    __jv_next_len;
99                    [$($acc,)* $pool_head];
100                    [$($pool_tail),*];
101                    $($tail),*
102                )
103            })
104            .collect::<::std::vec::Vec<_>>()
105    }};
106}

108#[cfg(test)]
109mod tests {
110    use super::*;
111
112    #[test]
113    fn test_1d() {
114        let v = jagged_vec![3; 0];
115        assert_eq!(v, vec![0, 0, 0]);
116    }
117
118    #[test]
119    fn test_1d_empty() {
120        let v: Vec<i32> = jagged_vec![0; 0];
121        assert_eq!(v, Vec::<i32>::new());
122    }
123
124    #[test]
125    fn test_2d_triangular() {
126        let v = jagged_vec![3; |i| i + 1; 0];
127        assert_eq!(v, vec![vec![0], vec![0, 0], vec![0, 0, 0]]);
128    }
129
130    #[test]
131    fn test_3d_skew() {
132        let n = 3;
133        let v = jagged_vec![n; |i| n - i, |i, j| n - i - j; 0];
134        assert_eq!(v, vec![
135            vec![vec![0, 0, 0], vec![0, 0], vec![0]],
136            vec![vec![0, 0], vec![0]],
137            vec![vec![0]],
138        ]);
139    }
140
141    #[test]
142    fn test_init_evaluated_once() {
143        use std::cell::Cell;
144        let calls = Cell::new(0);
145        let make = || {
146            calls.set(calls.get() + 1);
147            0
148        };
149        let v = jagged_vec![3; |i| i + 1; make()];
150        assert_eq!(calls.get(), 1);
151        assert_eq!(v, vec![vec![0], vec![0, 0], vec![0, 0, 0]]);
152    }
153
154    #[test]
155    fn test_non_copy_init() {
156        let v = jagged_vec![2; |i| i + 1; String::from("x")];
157        assert_eq!(v, vec![vec!["x".to_string()], vec!["x".into(), "x".into()]]);
158    }
159}
