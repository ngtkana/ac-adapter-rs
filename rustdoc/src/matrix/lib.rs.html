1//! 行列の積・べき乗、線形漸化式の行列表現（コンパニオン行列）。
2//!
3//! 要素の型は加減乗算そのものではなく、積和代入 $a \leftarrow a + bc$（[`Scalar::fma_assign`]）のみで
4//! 抽象化している。通常の数値環だけでなく、（最小, +）などの半環でも「積和代入」を適切に定義すれば
5//! 同じコードで行列積・べき乗が計算できる。行列積は積和代入を $O(hwp)$ 回適用して計算し、
6//! べき乗は繰り返し二乗法により $O(\log n)$ 回の行列積に帰着する。
7//!
8//! # 仕様
9//!
10//! - 型: `Matrix<T>`（`items: Vec<T>` を行優先（row-major）で格納、`width: usize` は列数）
11//! - [`Scalar`][]: 零元 `ZERO`、単位元 `ONE`、積和代入 `fma_assign(a, b, c)`（$a \leftarrow a + bc$）
12//! - 生成: [`Matrix::zeros`]、[`Matrix::identity`]、[`Matrix::companion`]
13//! - 演算: `&Matrix * &Matrix`（行列積）、[`Matrix::pow`]、[`Matrix::mul_pow_assign`]
14//! - 添字アクセス: `matrix[i]`（$i$ 行目のスライス）、`matrix[(i, j)]`（成分）
15//!
16//! # 例
17//!
18//! ```
19//! use matrix::Matrix;
20//! use matrix::Scalar;
21//!
22//! #[derive(Clone)]
23//! struct Int(i64);
24//! impl Scalar for Int {
25//!     const ZERO: Self = Self(0);
26//!     const ONE: Self = Self(1);
27//!     fn fma_assign(a: &mut Self, b: &Self, c: &Self) {
28//!         a.0 += b.0 * c.0;
29//!     }
30//! }
31//!
32//! let a = Matrix {
33//!     items: vec![Int(1), Int(2), Int(3), Int(4)],
34//!     width: 2,
35//! };
36//! let a2 = a.clone().pow(2); // [[1,2],[3,4]]^2
37//! assert_eq!(a2.items.iter().map(|x| x.0).collect::<Vec<_>>(), vec![7, 10, 15, 22]);
38//! ```
39//!
40//! # 計算量
41//!
42//! $h, w, p$ を行列の行数・共通次元・列数、$n$ をべき指数とする。
43//!
44//! - 行列積: $O(hwp)$
45//! - [`Matrix::pow`]、[`Matrix::mul_pow_assign`][]: $O(w^3 \log n)$（正方行列の場合）
46//! - [`Matrix::companion`][]: $O(k)$（$k$ は次数）
47

50/// 行列の要素が満たすべき演算：零元・単位元・積和代入。
51///
52/// 加算・乗算を個別に要求せず、積和代入 $a \leftarrow a + bc$（[`fma_assign`](Self::fma_assign)）のみを
53/// 要求する。これにより通常の数値環だけでなく、（最小, +）などの半環にも適用できる。
54pub trait Scalar {
55    /// 零元 $0$（加算の単位元）。
56    const ZERO: Self;
57    /// 単位元 $1$（乗算の単位元）。単位行列の対角成分に使う。
58    const ONE: Self;
59
60    /// 積和代入 $a \leftarrow a + bc$ を行う。
61    fn fma_assign(a: &mut Self, b: &Self, c: &Self);
62}
63
64/// 行優先（row-major）で格納した行列。
65///
66/// `items[i * width + j]` が $(i, j)$ 成分を表す。
67///
68/// # 例
69///
70/// ```
71/// use matrix::Matrix;
72/// let m = Matrix {
73///     items: vec![1, 2, 3, 4, 5, 6],
74///     width: 3,
75/// };
76/// assert_eq!(m.height(), 2);
77/// assert_eq!(m[(1, 2)], 6);
78/// ```
79#[derive(Clone)]
80pub struct Matrix<T> {
81    /// 成分（行優先）。長さは `width` の倍数。
82    pub items: Vec<T>,
83    /// 列数 $w$。
84    pub width: usize,
85}
86
87impl<T> Matrix<T> {
88    /// 行数 $h$ を返す（$h = $ `items.len() / width`）。
89    ///
90    /// # 例
91    ///
92    /// ```
93    /// use matrix::Matrix;
94    /// let m = Matrix {
95    ///     items: vec![0; 6],
96    ///     width: 3,
97    /// };
98    /// assert_eq!(m.height(), 2);
99    /// ```
100    pub fn height(&self) -> usize {
101        self.items.len() / self.width
102    }
103}
104
105impl<T: Scalar> Matrix<T> {
106    /// 零行列 $O_{h \times w}$（全成分が [`Scalar::ZERO`]）を生成する。
107    ///
108    /// # 例
109    ///
110    /// ```
111    /// # use matrix::Scalar;
112    /// # #[derive(Clone, Debug, PartialEq)]
113    /// # struct N(i64);
114    /// # impl Scalar for N {
115    /// #     const ZERO: Self = Self(0);
116    /// #     const ONE: Self = Self(1);
117    /// #     fn fma_assign(a: &mut Self, b: &Self, c: &Self) {
118    /// #         a.0 += b.0 * c.0;
119    /// #     }
120    /// # }
121    /// use matrix::Matrix;
122    /// let m = Matrix::<N>::zeros(2, 3);
123    /// assert_eq!(m.items, vec![N(0); 6]);
124    /// assert_eq!(m.width, 3);
125    /// ```
126    pub fn zeros(height: usize, width: usize) -> Self {
127        Self {
128            items: (0..height * width).map(|_| T::ZERO).collect::<Vec<_>>(),
129            width,
130        }
131    }
132
133    /// $n \times n$ の単位行列 $I_n$ を生成する。対角成分が [`Scalar::ONE`]、それ以外が
134    /// [`Scalar::ZERO`]。
135    ///
136    /// # 例
137    ///
138    /// ```
139    /// # use matrix::Scalar;
140    /// # #[derive(Clone, Debug, PartialEq)]
141    /// # struct N(i64);
142    /// # impl Scalar for N {
143    /// #     const ZERO: Self = Self(0);
144    /// #     const ONE: Self = Self(1);
145    /// #     fn fma_assign(a: &mut Self, b: &Self, c: &Self) {
146    /// #         a.0 += b.0 * c.0;
147    /// #     }
148    /// # }
149    /// use matrix::Matrix;
150    /// let m = Matrix::<N>::identity(2);
151    /// assert_eq!(m.items, vec![N(1), N(0), N(0), N(1)]);
152    /// ```
153    pub fn identity(size: usize) -> Self {
154        let mut result = Self::zeros(size, size);
155        for i in 0..size {
156            result[(i, i)] = T::ONE;
157        }
158        result
159    }
160
161    /// 行列のべき乗 $A^n$ を繰り返し二乗法で計算する。
162    ///
163    /// `self` は正方行列（`width == height()`）である必要がある。
164    ///
165    /// # 例
166    ///
167    /// ```
168    /// # use matrix::Scalar;
169    /// # #[derive(Clone, Debug, PartialEq)]
170    /// # struct N(i64);
171    /// # impl Scalar for N {
172    /// #     const ZERO: Self = Self(0);
173    /// #     const ONE: Self = Self(1);
174    /// #     fn fma_assign(a: &mut Self, b: &Self, c: &Self) {
175    /// #         a.0 += b.0 * c.0;
176    /// #     }
177    /// # }
178    /// use matrix::Matrix;
179    /// let a = Matrix {
180    ///     items: vec![N(1), N(2), N(3), N(4)],
181    ///     width: 2,
182    /// };
183    /// let a2 = a.pow(2); // [[1,2],[3,4]]^2
184    /// assert_eq!(a2.items, vec![N(7), N(10), N(15), N(22)]);
185    /// ```
186    ///
187    /// # 計算量
188    ///
189    /// $O(w^3 \log n)$（$w$ は行列の幅、$n$ は指数）
190    pub fn pow(self, exp: u64) -> Self {
191        let mut result = Self::identity(self.width);
192        result.mul_pow_assign(self, exp);
193        result
194    }
195
196    /// $\mathit{self} \leftarrow \mathit{self} \times \mathit{other}^n$ を繰り返し二乗法で計算する。
197    ///
198    /// `other` を $n$ 乗する過程で二乗した中間結果を `self` へ左から掛けていくことで、
199    /// `other` の明示的なべき乗計算を経ずに済ませる（[`pow`](Self::pow) は本メソッドを単位行列に
200    /// 適用したもの）。$n = 0$ のとき `self` は変化しない。
201    ///
202    /// # 例
203    ///
204    /// ```
205    /// # use matrix::Scalar;
206    /// # #[derive(Clone, Debug, PartialEq)]
207    /// # struct N(i64);
208    /// # impl Scalar for N {
209    /// #     const ZERO: Self = Self(0);
210    /// #     const ONE: Self = Self(1);
211    /// #     fn fma_assign(a: &mut Self, b: &Self, c: &Self) {
212    /// #         a.0 += b.0 * c.0;
213    /// #     }
214    /// # }
215    /// use matrix::Matrix;
216    /// let mut m = Matrix::<N>::identity(2);
217    /// let a = Matrix {
218    ///     items: vec![N(1), N(2), N(3), N(4)],
219    ///     width: 2,
220    /// };
221    /// m.mul_pow_assign(a, 2); // m <- I * [[1,2],[3,4]]^2
222    /// assert_eq!(m.items, vec![N(7), N(10), N(15), N(22)]);
223    /// ```
224    pub fn mul_pow_assign(&mut self, mut other: Self, mut exp: u64) {
225        if exp == 0 {
226            return;
227        }
228        while exp != 1 {
229            if exp & 1 == 1 {
230                *self = &*self * &other;
231            }
232            other = &other * &other;
233            exp >>= 1;
234        }
235        *self = &*self * &other;
236    }
237
238    /// 線形漸化式のコンパニオン行列を構成する。
239    ///
240    /// 係数 $\mathit{bottom} = (c_0, \ldots, c_{k-1})$ に対し、$k \times k$ 行列 $M$ を
241    /// $M_{i, i-1} = 1$（$1 \le i < k$）、$M_{i, k-1} = c_i$（$0 \le i < k$）、それ以外を $0$ として
242    /// 構成する。連続する $k$ 項からなる行ベクトル $u_n = (a_n, a_{n+1}, \ldots, a_{n+k-1})$ が
243    /// 漸化式 $a_{n+k} = \sum_{i=0}^{k-1} c_i a_{n+i}$ を満たすとき $u_n M = u_{n+1}$ が成り立つため、
244    /// $u_0 M^n$（[`pow`](Self::pow) 参照）の第 $0$ 成分から $a_n$ が求まる。
245    ///
246    /// # 例
247    ///
248    /// ```
249    /// # use matrix::Scalar;
250    /// # #[derive(Clone, Debug, PartialEq)]
251    /// # struct N(i64);
252    /// # impl Scalar for N {
253    /// #     const ZERO: Self = Self(0);
254    /// #     const ONE: Self = Self(1);
255    /// #     fn fma_assign(a: &mut Self, b: &Self, c: &Self) {
256    /// #         a.0 += b.0 * c.0;
257    /// #     }
258    /// # }
259    /// use matrix::Matrix;
260    /// // a_{n+2} = a_n + a_{n+1}（フィボナッチ）
261    /// let m = Matrix::companion(vec![N(1), N(1)]);
262    /// assert_eq!(m.items, vec![N(0), N(1), N(1), N(1)]); // [[0,1],[1,1]]
263    /// ```
264    ///
265    /// # Panics
266    ///
267    /// `bottom` が空のとき panic する。
268    pub fn companion<I>(bottom: I) -> Self
269    where
270        I: IntoIterator<Item = T>,
271        I::IntoIter: ExactSizeIterator,
272    {
273        let mut iter = bottom.into_iter();
274        let size = iter.len();
275        assert!(size > 0);
276        let mut result = Self::zeros(size, size);
277        for i in 1..size {
278            result[(i, i - 1)] = T::ONE;
279        }
280        for i in 0..size {
281            result[(i, size - 1)] = iter.next().unwrap();
282        }
283        assert!(iter.next().is_none());
284        result
285    }
286}

316/// 行列積 $C = AB$（$C_{ik} = \sum_j A_{ij} B_{jk}$）。積和は [`Scalar::fma_assign`] で計算する。
317///
318/// `self.width == rhs.height()` を満たす必要がある。
319///
320/// # 例
321///
322/// ```
323/// # use matrix::Scalar;
324/// # #[derive(Clone, Debug, PartialEq)]
325/// # struct N(i64);
326/// # impl Scalar for N {
327/// #     const ZERO: Self = Self(0);
328/// #     const ONE: Self = Self(1);
329/// #     fn fma_assign(a: &mut Self, b: &Self, c: &Self) {
330/// #         a.0 += b.0 * c.0;
331/// #     }
332/// # }
333/// use matrix::Matrix;
334/// let a = Matrix {
335///     items: vec![N(1), N(2), N(3), N(4)],
336///     width: 2,
337/// };
338/// let b = Matrix {
339///     items: vec![N(5), N(6), N(7), N(8)],
340///     width: 2,
341/// };
342/// let c = &a * &b;
343/// assert_eq!(c.items, vec![N(19), N(22), N(43), N(50)]);
344/// ```
345///
346/// # 計算量
347///
348/// $O(hwp)$（$h$: `self` の行数、$w$: `self.width`、$p$: `rhs.width`）
349impl<'a, T: Scalar> Mul<&'a Matrix<T>> for &'a Matrix<T> {
350    type Output = Matrix<T>;
351
352    fn mul(self, rhs: &'a Matrix<T>) -> Self::Output {
353        assert_eq!(self.width * rhs.width, rhs.items.len());
354        let mut result = Matrix::zeros(self.height(), rhs.width);
355        for i in 0..self.height() {
356            for j in 0..self.width {
357                for k in 0..rhs.width {
358                    T::fma_assign(&mut result[(i, k)], &self[(i, j)], &rhs[(j, k)]);
359                }
360            }
361        }
362        result
363    }
364}
