1//! モノイドの作用・積を繰り返し二乗法で $O(\log n)$ 回の演算に落とす二分累乗法。
2//!
3//! 指数 $n$ を二進展開し、上位ビットから「二乗して、そのビットが $1$ なら合成する」
4//! ことを繰り返すと、$n$ 回分の合成が $O(\log n)$ 回の二乗・合成で計算できる。
5//! 型 `T` の演算（二乗 `square` / 積 `mul`）と型 `U` への作用 `apply` を
6//! クロージャで渡すことで、行列累乗やモノイド作用など任意の演算に流用できる。
7//!
8//! # 仕様
9//!
10//! - [`Pow`]: 指数として使える型（符号なし整数）を表すトレイト
11//! - [`operator_binary`]: $a$ を $n$ 回 $x$ に作用させた結果 $a^n(x)$ を返す
12//! - [`value_binary`]: モノイドの積 $a^n$ を返す（単位元 `identity` 込み）
13//!
14//! # 例
15//!
16//! ```
17//! use binary::value_binary;
18//!
19//! // 3^5 = 243 を素朴な乗算の繰り返し二乗法で計算する
20//! let result = value_binary(3, 5_u32, 1, |&a, &b| a * b);
21//! assert_eq!(result, 243);
22//! ```
23//!
24//! # 計算量
25//!
26//! - [`operator_binary`], [`value_binary`]: `square`/`mul` の呼び出し $O(\log n)$ 回
27
28/// 二分累乗法の指数として使える符号なし整数型を表すトレイト。
29pub trait Pow {
30    /// $x \neq 0$
31    fn is_nonzero(&self) -> bool;
32    /// $x \neq 1$
33    fn is_nonone(&self) -> bool;
34    /// $x$ が奇数か
35    fn is_odd(&self) -> bool;
36    /// $x \mathrel{{/}{=}} 2$（右シフト）
37    fn shr1(&mut self);
38}
39
40/// $a$ を $x$ に $n$ 回作用させた結果 $a^n(x)$ を、$O(\log n)$ 回の `square`/`apply` で計算する。
41///
42/// `square` は $T$ 上の二乗、`apply` は $T$ の $U$ への作用（$a$ を $x$ に $1$ 回作用させる操作）。
43///
44/// # 例
45///
46/// ```
47/// use binary::operator_binary;
48///
49/// let a = 2;
50/// let n = 5_u32; // `i32` はコンパイルエラー
51/// let x = 42;
52/// let result = operator_binary(a, n, x, |&i| i * i, |&i, j| i * j);
53/// assert_eq!(result, 32 * x); // 2^5 = 32 を x = 42 に作用（乗算）
54/// ```
55pub fn operator_binary<T, U>(
56    mut a: T,
57    mut n: impl Pow,
58    mut x: U,
59    mut square: impl FnMut(&T) -> T,
60    mut apply: impl FnMut(&T, U) -> U,
61) -> U {
62    if n.is_nonzero() {
63        while n.is_nonone() {
64            if n.is_odd() {
65                x = apply(&a, x);
66            }
67            a = square(&a);
68            n.shr1();
69        }
70        x = apply(&a, x);
71    }
72    x
73}

75/// モノイドの積 $a^n$ を、単位元 `identity` と積 `mul` から $O(\log n)$ 回の `mul` で計算する。
76///
77/// # 例
78///
79/// ```
80/// use binary::value_binary;
81///
82/// let a = 3;
83/// let n = 5_u32; // `i32` はコンパイルエラー
84/// let result = value_binary(a, n, 1, |&i, &j| i * j);
85/// assert_eq!(result, 243); // 3^5 = 243
86/// ```
87pub fn value_binary<T>(
88    mut a: T,
89    mut n: impl Pow,
90    identity: T,
91    mut mul: impl FnMut(&T, &T) -> T,
92) -> T {
93    let mut ans = identity;
94    if n.is_nonzero() {
95        while n.is_nonone() {
96            if n.is_odd() {
97                ans = mul(&a, &ans);
98            }
99            a = mul(&a, &a);
100            n.shr1();
101        }
102        ans = mul(&a, &ans);
103    }
104    ans
105}

107macro_rules! impl_pow {
108    ($($T:ty),* $(,)?) => {$(
109        impl Pow for $T {
110            fn is_nonzero(&self) -> bool {
111                *self != 0
112            }
113            fn is_nonone(&self) -> bool {
114                *self != 1
115            }
116            fn is_odd(&self) -> bool {
117                self & 1 == 1
118            }
119            fn shr1(&mut self) {
120                *self >>= 1;
121            }
122        }
123    )*}
124}
