1//! 全方位木DP（rerooting）。各頂点を根としたときの集約値をまとめて計算する。
2//!
3//! 森を頂点集合 $V$、モノイド $(F, \mathrm{mul}, \mathrm{identity})$ の要素とみなし、
4//! 「頂点 $v$ をルートとして森 $F$ に接続し木にする」演算 $\mathrm{up}: F \times V \to F$ を追加で与える。
5//! 通常の部分木DP（子の集約値を `mul` で畳み込み `up` でルートに接続する）を根から葉・葉から根の
6//! 両方向に行うことで、「頂点 $v$ を根とする部分木の集約値」だけでなく「$v$ の親側を含めた
7//! 残り全体の集約値」も $O(n)$ 回の演算で求まる。`mul` は可換でなくてよく、各頂点で子を
8//! 元の順序で `mul` した前後の累積（prefix/suffix）を使って「自分を除いた残り」を計算する。
9//!
10//! # 仕様
11//!
12//! [`Op`] トレイトで演算を定義する。
13//!
14//! - `Op::up`: $F \times V \to F$。森に頂点 $v$ をルートとして接続する
15//! - `Op::mul`: $F \times F \to F$。森どうしの結合（結合律を満たせばよく、可換律は不要）
16//! - `Op::identity`: 空の森
17//!
18//! [`two_way_tree_fold`] は木（頂点数 $n$）を根から葉への有向グラフとして表す隣接リスト
19//! `g: &[Vec<usize>]`（`g[i]` は $i$ の子）と、親が子より前に来るトポロジカル順 `sorted` を
20//! 受け取り、[`TwoWayTreeFoldResult`] を返す。
21//!
22//! - `branch[i]`: 頂点 $i$ を根とする部分木の集約値
23//! - `lower[i]`: 頂点 $i$ の子部分木をすべて `mul` で結合した値（`branch[i] = up(lower[i], i)`）
24//! - `upper[i]`: 頂点 $i$ を除いた残り全体を $i$ に接続した集約値。根では `identity`
25//!
26//! # 例
27//!
28//! ```
29//! use tree_fold::Op;
30//! use tree_fold::two_way_tree_fold;
31//!
32//! struct SubtreeSize;
33//! impl Op for SubtreeSize {
34//!     type Value = usize;
35//!     fn up(&self, value: &usize, _root: usize) -> usize {
36//!         value + 1
37//!     }
38//!     fn mul(&self, lhs: &usize, rhs: &usize) -> usize {
39//!         lhs + rhs
40//!     }
41//!     fn identity(&self) -> usize {
42//!         0
43//!     }
44//! }
45//!
46//! // 木: 0 -> 1, 0 -> 2, 1 -> 3
47//! let g = vec![vec![1, 2], vec![3], vec![], vec![]];
48//! let sorted = vec![0, 1, 2, 3];
49//! let result = two_way_tree_fold(&SubtreeSize, &g, &sorted);
50//! assert_eq!(result.branch, vec![4, 2, 1, 1]); // 各頂点を根とする部分木のサイズ
51//! assert_eq!(result.upper, vec![0, 2, 3, 3]); // 各頂点から見た残り全体のサイズ（根は 0）
52//! ```
53//!
54//! # 計算量
55//!
56//! - `two_way_tree_fold`: $O(n)$ 回の `up`/`mul` 呼び出し（$n$ は頂点数）
57
58/// 全方位木DPの演算。モノイド $(F, \mathrm{mul}, \mathrm{identity})$ と、
59/// 頂点 $v$ をルートとして森 $F$ に接続する演算 $\mathrm{up}$ を定義する。
60pub trait Op: Sized {
61    /// モノイド $F$ の値の型。
62    type Value: Clone;
63
64    /// 森 `value` に頂点 `root` をルートとして接続する: $F \times V \to F$。
65    fn up(&self, value: &Self::Value, root: usize) -> Self::Value;
66
67    /// 森どうしを結合する: $F \times F \to F$（結合律を満たせばよく、可換律は不要）。
68    fn mul(&self, lhs: &Self::Value, rhs: &Self::Value) -> Self::Value;
69
70    /// 空の森を返す: $\mathrm{identity} \in F$。
71    fn identity(&self) -> Self::Value;
72
73    /// [`two_way_tree_fold`] を呼び出す。詳細はそちらを参照。
74    fn two_way_tree_fold(
75        &self,
76        g: &[Vec<usize>],
77        sorted: &[usize],
78    ) -> TwoWayTreeFoldResult<Self::Value> {
79        two_way_tree_fold(self, g, sorted)
80    }
81}

83/// [`two_way_tree_fold`] の返り値。
84pub struct TwoWayTreeFoldResult<T> {
85    /// `upper[i]`: 頂点 $i$ を除いた残り全体を $i$ に接続した集約値。根では `identity`。
86    pub upper: Vec<T>,
87    /// `lower[i]`: 頂点 $i$ の子部分木をすべて `mul` で結合した値。
88    pub lower: Vec<T>,
89    /// `branch[i]`: 頂点 $i$ を根とする部分木の集約値（`up(lower[i], i)`）。
90    pub branch: Vec<T>,
91}
92
93/// 全方位木DPを行う。詳細はモジュールレベルドキュメントの `# 仕様` を参照。
94pub fn two_way_tree_fold<O: Op>(
95    o: &O,
96    g: &[Vec<usize>],
97    sorted: &[usize],
98) -> TwoWayTreeFoldResult<O::Value> {
99    let n = g.len();
100    let mut lower = vec![o.identity(); n];
101    let mut branch = vec![o.identity(); n];
102    for &i in sorted.iter().rev() {
103        for &j in &g[i] {
104            lower[i] = o.mul(&lower[i], &branch[j]);
105        }
106        branch[i] = o.up(&lower[i], i);
107    }
108    let mut upper = vec![o.identity(); n];
109    for &i in sorted {
110        let mut suffix = upper[i].clone();
111        for &j in g[i].iter().rev() {
112            upper[j] = suffix.clone();
113            suffix = o.mul(&branch[j], &suffix);
114        }
115        let mut prefix = o.identity();
116        for &j in &g[i] {
117            upper[j] = o.up(&o.mul(&upper[j], &prefix), i);
118            prefix = o.mul(&prefix, &branch[j]);
119        }
120    }
121    TwoWayTreeFoldResult {
122        upper,
123        lower,
124        branch,
125    }
126}

128#[cfg(test)]
129mod tests {
130    use super::*;
131    use rand::rngs::StdRng;
132    use rand::Rng;
133    use rand::SeedableRng;
134    use std::collections::BinaryHeap;
135
136    struct GenratedTree {
137        g: Vec<Vec<usize>>,
138        sorted: Vec<usize>,
139        parent: Vec<usize>,
140    }
141
142    fn gen_tree(rng: &mut StdRng, n: usize) -> GenratedTree {
143        let mut g = vec![Vec::new(); n];
144        let rev_prufer = (0..n - 1)
145            .map(|i| if i == n - 2 { 0 } else { rng.gen_range(0..n) })
146            .collect::<Vec<_>>();
147        let mut count = vec![0; n];
148        for &y in &rev_prufer {
149            count[y] += 1;
150        }
151        let mut heap = (0..n).filter(|&y| count[y] == 0).collect::<BinaryHeap<_>>();
152        for &y in &rev_prufer {
153            let x = heap.pop().unwrap();
154            g[x].push(y);
155            g[y].push(x);
156            count[y] -= 1;
157            if count[y] == 0 {
158                heap.push(y);
159            }
160        }
161        let root = rng.gen_range(0..n);
162        let mut stack = vec![root];
163        let mut parent = vec![usize::MAX; n];
164        let mut sorted = Vec::new();
165        parent[root] = root;
166        while let Some(i) = stack.pop() {
167            sorted.push(i);
168            g[i].retain(|&j| parent[i] != j);
169            for &j in &g[i] {
170                stack.push(j);
171                parent[j] = i;
172            }
173        }
174        GenratedTree { g, sorted, parent }
175    }

177    #[test]
178    fn test_subtree() {
179        struct O;
180        #[derive(Clone, PartialEq)]
181        struct Tree {
182            root: usize,
183            child: Vec<Self>,
184        }
185        impl std::fmt::Debug for Tree {
186            fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
187                f.debug_tuple("")
188                    .field(&self.root)
189                    .field(&self.child)
190                    .finish()
191            }
192        }
193        impl Op for O {
194            type Value = Vec<Tree>;
195
196            fn up(&self, value: &Self::Value, root: usize) -> Self::Value {
197                vec![Tree {
198                    root,
199                    child: value.clone(),
200                }]
201            }
202
203            fn identity(&self) -> Self::Value {
204                Vec::new()
205            }
206
207            fn mul(&self, lhs: &Self::Value, rhs: &Self::Value) -> Self::Value {
208                lhs.iter().cloned().chain(rhs.iter().cloned()).collect()
209            }
210        }
211
212        fn subtree(i: usize, p: usize, g: &[Vec<usize>]) -> Tree {
213            Tree {
214                root: i,
215                child: match g[i].iter().position(|&j| j == p) {
216                    None => g[i].iter().map(|&j| subtree(j, i, g)).collect(),
217                    Some(e) => g[i][e + 1..]
218                        .iter()
219                        .chain(&g[i][..e])
220                        .map(|&j| subtree(j, i, g))
221                        .collect(),
222                },
223            }
224        }
225
226        let mut rng = StdRng::seed_from_u64(42);
227        for _ in 0..100 {
228            let n = rng.gen_range(1..=30);
229            let GenratedTree {
230                g, sorted, parent, ..
231            } = gen_tree(&mut rng, n);
232            let TwoWayTreeFoldResult {
233                lower,
234                branch,
235                upper,
236            } = O.two_way_tree_fold(&g, &sorted);
237            let mut g1 = g.clone();
238            for (i, &p) in parent.iter().enumerate() {
239                if i != p {
240                    g1[i].push(p);
241                }
242            }
243            for (i, &p) in parent.iter().enumerate() {
244                assert_eq!(
245                    lower[i],
246                    g[i].iter().map(|&j| subtree(j, i, &g)).collect::<Vec<_>>()
247                );
248                assert_eq!(branch[i], vec![subtree(i, p, &g)]);
249                assert_eq!(
250                    upper[i],
251                    if i == p { Vec::new() } else { vec![subtree(p, i, &g1)] }
252                );
253            }
254        }
255    }
