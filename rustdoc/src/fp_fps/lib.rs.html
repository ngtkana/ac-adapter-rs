3use fp::Fp;
4use fp::fp_new;
5use fp_fft::fft;
6use fp_fft::ifft;

8/// $f^{-1} \bmod x^m$ を計算する。
9///
10/// # 計算量
11///
12/// $m$ が $2$ べきのとき、 $10 \mathcal{F}_m$
13///
14/// # 例
15///
16/// ```
17/// use fp::fp_new;
18/// use fp_fps::fps_inv;
19/// const P: u64 = 998_244_353;
20/// let f = [fp_new::<P>(1), fp_new::<P>(2)];
21/// let g = fps_inv(&f, 2);
22/// assert_eq!(g[0], fp_new::<P>(1)); // (1+2x)^{-1} の 0 次項 = 1
23/// assert_eq!(g[1], fp_new::<P>(998244351)); // (1+2x)^{-1} の 1 次項 = -2
24/// ```
25pub fn fps_inv<const P: u64>(f: &[Fp<P>], precision: usize) -> Vec<Fp<P>> {
26    let fft_len_max = precision.next_power_of_two();
27    let mut g = vec![fp_new(0); precision];
28    g[0] = f[0].inv();
29    let mut h = vec![fp_new(0); fft_len_max];
30    let mut g_fft = vec![fp_new(0); fft_len_max];
31    let mut fft_len = 2;
32    while fft_len <= fft_len_max {
33        if fft_len < f.len() {
34            h[..fft_len].copy_from_slice(&f[..fft_len]);
35        } else {
36            h[..f.len()].copy_from_slice(&f[..f.len()]);
37            h[f.len()..fft_len].fill(fp_new(0));
38        }
39        g_fft[..fft_len / 2].copy_from_slice(&g[..fft_len / 2]);
40        fft(&mut h[..fft_len]);
41        fft(&mut g_fft[..fft_len]);
42        for i in 0..fft_len {
43            h[i] = fp_new(1) - h[i] * g_fft[i];
44        }
45        ifft(&mut h[..fft_len]);
46        h[..fft_len / 2].fill(fp_new(0));
47        fft(&mut h[..fft_len]);
48        for i in 0..fft_len {
49            h[i] *= g_fft[i];
50        }
51        ifft(&mut h[..fft_len]);
52        g[fft_len / 2..fft_len.min(precision)]
53            .copy_from_slice(&h[fft_len / 2..fft_len.min(precision)]);
54        fft_len *= 2;
55    }
56    g
57}

59/// 多項式 $a$ と $b$ を乗算。
60///
61/// * 出力の trailing $0$ は全て削除されれる
62/// * 特に、$0$ 多項式は空が返される
63///
64/// 計算量 $O(n \log n)$。
65///
66/// # 例
67///
68/// ```
69/// use fp::fp_new;
70/// use fp_fps::poly_mul;
71/// const P: u64 = 998_244_353;
72/// let a = [fp_new(1), fp_new(2)]; // 1 + 2x
73/// let b = [fp_new(3), fp_new(4)]; // 3 + 4x
74/// let c = poly_mul::<P>(a.to_vec(), b.to_vec());
75/// assert_eq!(c.as_slice(), [fp_new(3), fp_new(10), fp_new(8)]);
76/// ```
77pub fn poly_mul<const P: u64>(mut a: Vec<Fp<P>>, mut b: Vec<Fp<P>>) -> Vec<Fp<P>> {
78    if a.is_empty() {
79        return b;
80    }
81    if b.is_empty() {
82        return a;
83    }
84    let len = a.len() + b.len() - 1;
85    let fft_len = len.next_power_of_two() * 2;
86    a.resize(fft_len, fp_new(0));
87    b.resize(fft_len, fp_new(0));
88    fft(&mut a);
89    fft(&mut b);
90    for i in 0..fft_len {
91        a[i] *= b[i];
92    }
93    ifft(&mut a);
94    a.truncate(len);
95    a
96}

98/// 多項式除算。$a = bq + r$ を満たす $(q, r)$ を返す（$\deg(r) < \deg(b)$）。
99///
100/// 計算量 $O(n \log n)$。
101///
102/// # 例
103///
104/// ```
105/// use fp::fp_new;
106/// use fp_fps::poly_div_rem;
107/// const P: u64 = 998_244_353;
108/// let a = [fp_new(1), fp_new(0), fp_new(1)]; // 1 + x^2
109/// let b = [fp_new(1), fp_new(1)]; // 1 + x
110/// let (q, r) = poly_div_rem::<P>(a.to_vec(), b.to_vec());
111/// assert_eq!(q.as_slice(), &[-fp_new(1), fp_new(1)]); // -1 + x
112/// assert_eq!(r.as_slice(), &[fp_new(2)]); // 2
113/// ```
114pub fn poly_div_rem<const P: u64>(
115    mut a: Vec<Fp<P>>,
116    mut b: Vec<Fp<P>>,
117) -> (Vec<Fp<P>>, Vec<Fp<P>>) {
118    assert_ne!(*b.last().unwrap(), fp_new(0));
119    if a.len() < b.len() {
120        return (vec![], a);
121    }
122    let d = b.iter().position(|&b| b != fp_new(0)).unwrap();
123    a[d..].reverse();
124    b[d..].reverse();
125    let precision = a.len() - b.len() + 1;
126    let mut q = poly_mul(
127        a[d..a.len().min(d + precision)].to_vec(),
128        fps_inv(&b[d..], precision),
129    );
130    q.truncate(precision);
131    q.reverse();
132    a[d..].reverse();
133    b[d..].reverse();
134    let bq = poly_mul(b, q.clone());
135    for i in 0..bq.len() {
136        a[i] -= bq[i];
137    }
138    while a.pop_if(|&mut a| a == fp_new(0)).is_some() {}
139    (q, a)
140}

142/// 多項式 $f$ を複数点 $x_1, \ldots, x_n$ で評価。$[f(x_1), \ldots, f(x_n)]$ を返す。
143///
144/// 分割統治 + FFT、計算量 $O(n \log^2 n)$。
145///
146/// # 例
147///
148/// ```
149/// use fp::fp_new;
150/// use fp_fps::multipoint_evaluation;
151/// const P: u64 = 998_244_353;
152/// let f = [fp_new::<P>(1), fp_new::<P>(2)]; // 1 + 2x
153/// let points = [fp_new::<P>(0), fp_new::<P>(1)];
154/// let result = multipoint_evaluation(f.to_vec(), &points);
155/// assert_eq!(result[0], fp_new::<P>(1)); // f(0) = 1
156/// assert_eq!(result[1], fp_new::<P>(3)); // f(1) = 3
157/// ```
158pub fn multipoint_evaluation<const P: u64>(f: Vec<Fp<P>>, points: &[Fp<P>]) -> Vec<Fp<P>> {
159    let n = points.len();
160    let mut prod = vec![vec![]; n * 2];
161    for (prod, &point) in prod[n..].iter_mut().zip(points) {
162        *prod = vec![-point, fp_new(1)];
163    }
164    for i in (1..n).rev() {
165        prod[i] = poly_mul(prod[2 * i].clone(), prod[2 * i + 1].clone());
166    }
167    let mut rem = vec![vec![]; n * 2];
168    rem[1] = poly_div_rem(f, prod[1].clone()).1;
169    for i in 1..n {
170        rem[2 * i] = poly_div_rem(rem[i].clone(), prod[2 * i].clone()).1;
171        rem[2 * i + 1] = poly_div_rem(rem[i].clone(), prod[2 * i + 1].clone()).1;
172    }
173    rem[n..]
174        .iter()
175        .map(|ans| ans.first().copied().unwrap_or(fp_new(0)))
176        .collect()
177}
