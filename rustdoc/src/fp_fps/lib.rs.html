1//! 形式的冪級数（FPS）の演算。多項式の乗算・逆元・除算・多点評価を FFT で高速化する。
2//!
3//! 多項式の積は係数列の畳み込みに一致するため、[`fp_fft`] の FFT を用いて $O(n \log n)$ で計算できる。
4//! 逆元はニュートン法により精度を倍々に伸ばしながら FFT で更新することで $O(n \log n)$ を達成する。
5//! 除算・多点評価はこれらを組み合わせて構築する。
6//!
7//! # 仕様
8//!
9//! - [`poly_mul`][]: 多項式の積 $ab$
10//! - [`fps_inv`][]: 形式的冪級数の逆元 $f^{-1} \bmod x^m$（$f(0) \neq 0$ が前提）
11//! - [`poly_div_rem`][]: 多項式の除算 $a = bq + r$（$\deg r < \deg b$）
12//! - [`multipoint_evaluation`][]: 多項式の複数点評価 $f(x_1), \ldots, f(x_n)$
13//!
14//! # 例
15//!
16//! ```
17//! use fp::fp_new;
18//! use fp_fps::poly_mul;
19//! const P: u64 = 998_244_353;
20//! let a = [fp_new::<P>(1), fp_new::<P>(2)]; // 1 + 2x
21//! let b = [fp_new::<P>(3), fp_new::<P>(4)]; // 3 + 4x
22//! let c = poly_mul::<P>(a.to_vec(), b.to_vec());
23//! assert_eq!(c.as_slice(), [fp_new(3), fp_new(10), fp_new(8)]); // (1+2x)(3+4x) = 3+10x+8x^2
24//! ```
25//!
26//! # 計算量
27//!
28//! - [`poly_mul`][]: $O(n \log n)$
29//! - [`fps_inv`][]: $O(n \log n)$
30//! - [`poly_div_rem`][]: $O(n \log n)$
31//! - [`multipoint_evaluation`][]: $O(n \log^2 n)$
32
33use fp::Fp;
34use fp::fp_new;
35use fp_fft::fft;
36use fp_fft::ifft;

38/// 形式的冪級数 $f$ の逆元を $\bmod x^m$ で計算する。
39///
40/// ニュートン法で精度を倍々に伸ばす。精度 $k$ の逆元 $g_k$ から
41/// $g_{2k} = g_k (2 - f g_k) \bmod x^{2k}$ により精度 $2k$ の逆元を求める。
42///
43/// # 仕様
44///
45/// $f(0) \neq 0$ のとき、$fg \equiv 1 \pmod{x^m}$（$m = $ `precision`）を満たす
46/// 長さ $m$ の係数列 $g$ を返す。
47///
48/// # 例
49///
50/// ```
51/// use fp::fp_new;
52/// use fp_fps::fps_inv;
53/// const P: u64 = 998_244_353;
54/// let f = [fp_new::<P>(1), fp_new::<P>(2)];
55/// let g = fps_inv(&f, 2);
56/// assert_eq!(g[0], fp_new::<P>(1)); // (1+2x)^{-1} の 0 次項 = 1
57/// assert_eq!(g[1], fp_new::<P>(998244351)); // (1+2x)^{-1} の 1 次項 = -2
58/// ```
59///
60/// # 計算量
61///
62/// $O(n \log n)$（$n = m$）
63pub fn fps_inv<const P: u64>(f: &[Fp<P>], precision: usize) -> Vec<Fp<P>> {
64    let fft_len_max = precision.next_power_of_two();
65    let mut g = vec![fp_new(0); precision];
66    g[0] = f[0].inv();
67    let mut h = vec![fp_new(0); fft_len_max];
68    let mut g_fft = vec![fp_new(0); fft_len_max];
69    let mut fft_len = 2;
70    while fft_len <= fft_len_max {
71        if fft_len < f.len() {
72            h[..fft_len].copy_from_slice(&f[..fft_len]);
73        } else {
74            h[..f.len()].copy_from_slice(&f[..f.len()]);
75            h[f.len()..fft_len].fill(fp_new(0));
76        }
77        g_fft[..fft_len / 2].copy_from_slice(&g[..fft_len / 2]);
78        fft(&mut h[..fft_len]);
79        fft(&mut g_fft[..fft_len]);
80        for i in 0..fft_len {
81            h[i] = fp_new(1) - h[i] * g_fft[i];
82        }
83        ifft(&mut h[..fft_len]);
84        h[..fft_len / 2].fill(fp_new(0));
85        fft(&mut h[..fft_len]);
86        for i in 0..fft_len {
87            h[i] *= g_fft[i];
88        }
89        ifft(&mut h[..fft_len]);
90        g[fft_len / 2..fft_len.min(precision)]
91            .copy_from_slice(&h[fft_len / 2..fft_len.min(precision)]);
92        fft_len *= 2;
93    }
94    g
95}

97/// 多項式 $a$ と $b$ の積 $ab$ を計算する。
98///
99/// 係数列を FFT で周波数領域に写し、要素ごとの積を取ってから逆変換することで、
100/// 畳み込み（多項式の積）を $O(n \log n)$ で求める。
101///
102/// # 仕様
103///
104/// 出力の末尾の $0$ 係数はすべて取り除く。特に、$a$ または $b$ が $0$ 多項式なら空配列を返す。
105///
106/// # 例
107///
108/// ```
109/// use fp::fp_new;
110/// use fp_fps::poly_mul;
111/// const P: u64 = 998_244_353;
112/// let a = [fp_new(1), fp_new(2)]; // 1 + 2x
113/// let b = [fp_new(3), fp_new(4)]; // 3 + 4x
114/// let c = poly_mul::<P>(a.to_vec(), b.to_vec());
115/// assert_eq!(c.as_slice(), [fp_new(3), fp_new(10), fp_new(8)]); // (1+2x)(3+4x) = 3+10x+8x^2
116/// ```
117///
118/// # 計算量
119///
120/// $O(n \log n)$（$n = |a| + |b|$）
121pub fn poly_mul<const P: u64>(mut a: Vec<Fp<P>>, mut b: Vec<Fp<P>>) -> Vec<Fp<P>> {
122    if a.is_empty() {
123        return b;
124    }
125    if b.is_empty() {
126        return a;
127    }
128    let len = a.len() + b.len() - 1;
129    let fft_len = len.next_power_of_two();
130    a.resize(fft_len, fp_new(0));
131    b.resize(fft_len, fp_new(0));
132    fft(&mut a);
133    fft(&mut b);
134    for i in 0..fft_len {
135        a[i] *= b[i];
136    }
137    ifft(&mut a);
138    a.truncate(len);
139    a
140}

142/// 多項式除算。$a = bq + r$（$\deg(r) < \deg(b)$）を満たす $(q, r)$ を返す。
143///
144/// $a, b$ の係数列を反転すると、$q$ の反転は $b$ の反転の逆元と $a$ の反転の積の
145/// 先頭 $\deg(a) - \deg(b) + 1$ 項に一致する。これを [`fps_inv`], [`poly_mul`] で求め、
146/// 反転を戻してから $r = a - bq$ で余りを計算する。
147///
148/// # 仕様
149///
150/// $b$ の最高次係数は $0$ でないことが前提。
151///
152/// # 例
153///
154/// ```
155/// use fp::fp_new;
156/// use fp_fps::poly_div_rem;
157/// const P: u64 = 998_244_353;
158/// let a = [fp_new(1), fp_new(0), fp_new(1)]; // 1 + x^2
159/// let b = [fp_new(1), fp_new(1)]; // 1 + x
160/// let (q, r) = poly_div_rem::<P>(a.to_vec(), b.to_vec());
161/// assert_eq!(q.as_slice(), &[-fp_new(1), fp_new(1)]); // -1 + x
162/// assert_eq!(r.as_slice(), &[fp_new(2)]); // 2
163/// ```
164///
165/// # 計算量
166///
167/// $O(n \log n)$
168pub fn poly_div_rem<const P: u64>(
169    mut a: Vec<Fp<P>>,
170    mut b: Vec<Fp<P>>,
171) -> (Vec<Fp<P>>, Vec<Fp<P>>) {
172    assert_ne!(*b.last().unwrap(), fp_new(0));
173    if a.len() < b.len() {
174        return (vec![], a);
175    }
176    let d = b.iter().position(|&b| b != fp_new(0)).unwrap();
177    a[d..].reverse();
178    b[d..].reverse();
179    let precision = a.len() - b.len() + 1;
180    let mut q = poly_mul(
181        a[d..a.len().min(d + precision)].to_vec(),
182        fps_inv(&b[d..], precision),
183    );
184    q.truncate(precision);
185    q.reverse();
186    a[d..].reverse();
187    b[d..].reverse();
188    let bq = poly_mul(b, q.clone());
189    for i in 0..bq.len() {
190        a[i] -= bq[i];
191    }
192    while a.pop_if(|&mut a| a == fp_new(0)).is_some() {}
193    (q, a)
194}

196/// 多項式 $f$ を複数の点 $x_1, \ldots, x_n$ で評価する。
197///
198/// 分割統治で $\prod_i (x - x_i)$ を部分木ごとに構築し（葉から根へ）、
199/// 根から葉に向かって $f$ をその部分木の積多項式で割った余りを伝播させる
200/// （[`poly_div_rem`]）ことで、各葉で $f(x_i)$ を得る。
201///
202/// # 仕様
203///
204/// $[f(x_1), \ldots, f(x_n)]$ を返す。
205///
206/// # 例
207///
208/// ```
209/// use fp::fp_new;
210/// use fp_fps::multipoint_evaluation;
211/// const P: u64 = 998_244_353;
212/// let f = [fp_new::<P>(1), fp_new::<P>(2)]; // 1 + 2x
213/// let points = [fp_new::<P>(0), fp_new::<P>(1)];
214/// let result = multipoint_evaluation(f.to_vec(), &points);
215/// assert_eq!(result[0], fp_new::<P>(1)); // f(0) = 1
216/// assert_eq!(result[1], fp_new::<P>(3)); // f(1) = 3
217/// ```
218///
219/// # 計算量
220///
221/// $O(n \log^2 n)$
222pub fn multipoint_evaluation<const P: u64>(f: Vec<Fp<P>>, points: &[Fp<P>]) -> Vec<Fp<P>> {
223    let n = points.len();
224    let mut prod = vec![vec![]; n * 2];
225    for (prod, &point) in prod[n..].iter_mut().zip(points) {
226        *prod = vec![-point, fp_new(1)];
227    }
228    for i in (1..n).rev() {
229        prod[i] = poly_mul(prod[2 * i].clone(), prod[2 * i + 1].clone());
230    }
231    let mut rem = vec![vec![]; n * 2];
232    rem[1] = poly_div_rem(f, prod[1].clone()).1;
233    for i in 1..n {
234        rem[2 * i] = poly_div_rem(rem[i].clone(), prod[2 * i].clone()).1;
235        rem[2 * i + 1] = poly_div_rem(rem[i].clone(), prod[2 * i + 1].clone()).1;
236    }
237    rem[n..]
238        .iter()
239        .map(|ans| ans.first().copied().unwrap_or(fp_new(0)))
240        .collect()
241}
