1//! Hopcroft-Karp アルゴリズムによる二部グラフの最大マッチング
2//!
3//! # 仕様
4//!
5//! 有向グラフ $G = (V, E)$ が与えられたとき、$L = \\{ x \in V \mid \exists y \in V: (x, y) \in E \\}$ とし、
6//! 二部グラフ $(L, R, E)$ （$R = V \setminus L$）における**最大マッチング** $M \subseteq E$ を求める。
7//!
8//! マッチング $M$ は以下を満たす：
9//! - $M$ の各エッジは頂点を共有しない（独立エッジ集合）
10//! - 任意のエッジ $e \in E \setminus M$ に対し、$e$ を $M$ に追加すると独立エッジ集合でなくなる
11//!
12//! # 入出力
13//!
14//! - **入力**: グラフ $g: L \to \mathcal{P}(R)$ （隣接リスト表現）
15//!   - $g(i) = \\{ j : (i, j) \in E \\}$ （$i \in L$）
16//! - **出力**: 関数 $f: R \to L \cup \\{\infty\\}$ （マッチング $M = \\{ (f(j), j) : f(j) \neq \infty \\}$）
17//!   - $f(j) = i$ なら $(i, j) \in M$
18//!   - $f(j) = \infty$ なら $j$ はマッチされていない
19//!
20//! # 例
21//!
22//! ```
23//! # use bipartite_matching::bipartite_matching;
24//! let g = vec![vec![2], vec![2], vec![]];  // $L = \\{0, 1\\}$, $R = \\{2\\}$
25//! let f = bipartite_matching(&g);
26//! // 最大マッチング: $0 \to 2$ （$f(2) = 0$）
27//! assert_eq!(f[2], 0);
28//! ```
29//!
30//! # 計算量
31//!
32//! $O(\sqrt{V} \cdot E)$ （$V$ = ノード数、$E$ = エッジ数）
33
34use std::{collections::VecDeque, mem::replace};
35
36/// 二部グラフの最大マッチングを求める
37///
38/// Hopcroft-Karp アルゴリズムにより $O(\sqrt{V} \cdot E)$ で計算。
39///
40/// # 入力
41/// - `g`: グラフの隣接リスト。$g(i)$ = 左頂点 $i$ から到達可能な右側頂点集合
42///
43/// # 出力
44/// マッチング関数 $f: R \to L \cup \\{\infty\\}$
45/// - $f(j) = i$ ($i \neq \infty$) ⟹ 左頂点 $i$ が右頂点 $j$ とマッチング
46/// - $f(j) = \infty$ ⟹ 右頂点 $j$ はマッチされていない
47///
48/// # 例
49/// ```
50/// # use bipartite_matching::bipartite_matching;
51/// // 二部グラフ: $L = \\{0, 1\\}$, $R = \\{2\\}$
52/// // エッジ: $0 \to 2$, $1 \to 2$
53/// let g = vec![vec![2], vec![2], vec![]];
54/// let f = bipartite_matching(&g);
55/// // マッチング: $0 \to 2$ （$f(2) = 0$）
56/// assert_eq!(f[2], 0);
57/// ```
58pub fn bipartite_matching(g: &[Vec<usize>]) -> Vec<usize> {
59    let n = g.len();
60    let mut queue = VecDeque::new();
61    let mut f = vec![usize::MAX; n];
62    let mut label = vec![usize::MAX; n];
63    for x in (0..n).filter(|&i| !g[i].is_empty()) {
64        queue.push_back(x);
65        label[x] = 0;
66    }
67    loop {
68        let orig_count = queue.len();
69        while let Some(x) = queue.pop_front() {
70            for &y in &g[x] {
71                if label[y] == usize::MAX {
72                    label[y] = label[x] + 1;
73                    let z = f[y];
74                    if z != usize::MAX && label[z] == usize::MAX {
75                        label[z] = label[x] + 2;
76                        queue.push_back(z);
77                    }
78                }
79            }
80        }
81        for x in 0..n {
82            if label[x] == 0 && !primal(x, &mut label, g, &mut f) {
83                queue.push_back(x);
84            }
85        }
86        if orig_count == queue.len() || queue.is_empty() {
87            return f;
88        }
89        label.fill(usize::MAX);
90        for &x in &queue {
91            label[x] = 0;
92        }
93    }
94}

96/// 左頂点 $x$ から増加経路を DFS で探索
97///
98/// ラベル値を利用した「現在エッジ」構造：探索済みノードの再訪問を防ぐ。
99fn primal(x: usize, label: &mut [usize], g: &[Vec<usize>], f: &mut [usize]) -> bool {
100    let d = replace(&mut label[x], usize::MAX);
101    for &y in &g[x] {
102        if d + 1 == label[y] {
103            let z = f[y];
104            if z == usize::MAX || (d + 2 == label[z] && primal(z, label, g, f)) {
105                f[y] = x;
106                return true;
107            }
108        }
109    }
110    false
111}
