1use anyhow::Context;
2use anyhow::Result;
3use std::fs;
4use std::path::Path;
5use syn::Item;
6
7/// `lib.rs` を起点に、`mod foo;` 宣言を再帰的に `src/foo.rs` の中身で埋め込んだ `Item` 列を返す。
8///
9/// `foo/mod.rs` 形式のサブディレクトリは非対応（ac-adapter-rs の `libs/*` に存在しないため）。
10pub fn load_crate_items(lib_rs: &Path) -> Result<Vec<Item>> {
11    let dir = lib_rs
12        .parent()
13        .expect("lib.rs always has a parent directory");
14    let items = parse_file(lib_rs)?;
15    inline_items(items, dir)
16}
17
18fn parse_file(path: &Path) -> Result<Vec<Item>> {
19    let source =
20        fs::read_to_string(path).with_context(|| format!("failed to read {}", path.display()))?;
21    let file = syn::parse_file(&source)
22        .with_context(|| format!("failed to parse {} as Rust source", path.display()))?;
23    Ok(file.items)
24}
25
26fn inline_items(items: Vec<Item>, dir: &Path) -> Result<Vec<Item>> {
27    items
28        .into_iter()
29        .map(|item| inline_item(item, dir))
30        .collect()
31}
32
33fn inline_item(item: Item, dir: &Path) -> Result<Item> {
34    let Item::Mod(mut item_mod) = item else {
35        return Ok(item);
36    };
37    if item_mod.content.is_some() {
38        return Ok(Item::Mod(item_mod));
39    }
40    let sub_path = dir.join(format!("{}.rs", item_mod.ident));
41    let sub_items = parse_file(&sub_path)?;
42    let sub_items = inline_items(sub_items, dir)?;
43    item_mod.content = Some((syn::token::Brace::default(), sub_items));
44    item_mod.semi = None;
45    Ok(Item::Mod(item_mod))
46}
