1use cargo_metadata::DependencyKind;
2use cargo_metadata::MetadataCommand;

7#[derive(Clone, Debug, PartialEq, serde::Serialize)]
8struct CrateMetadata {
9    dependencies: Vec<String>,
10    tags: Vec<String>,
11    /// crate root doc comment の要約行（1行目）のプレーンテキスト。検索用。
12    description: Option<String>,
13    /// 要約行を HTML にレンダリングしたもの。表示用。
14    description_html: Option<String>,
15    /// crate root doc comment の要約行より後（本文）を HTML にレンダリングしたもの。
16    full: Option<String>,
17}
18
19fn main() {
20    let metadata = MetadataCommand::new().no_deps().exec().unwrap();
21    let project_root = metadata.workspace_root.clone();
22
23    let crate_metadatas: HashMap<String, CrateMetadata> = metadata
24        .packages
25        .iter()
26        .filter(|package| package.manifest_path.starts_with(project_root.join("libs")))
27        .map(|package| {
28            let dependencies = package
29                .dependencies
30                .iter()
31                .filter(|dep| dep.kind == DependencyKind::Normal && dep.path.is_some())
32                .map(|dep| dep.name.clone())
33                .collect();
34            let tags = package.keywords.clone();
35            let lib_rs_path: PathBuf = package
36                .manifest_path
37                .parent()
38                .unwrap()
39                .join("src")
40                .join("lib.rs")
41                .into();
42            let (description, description_html, full) = fs::read_to_string(&lib_rs_path)
43                .map(|content| parse_crate_docs(&content))
44                .unwrap_or((None, None, None));
45
46            (
47                package.name.clone(),
48                CrateMetadata {
49                    dependencies,
50                    tags,
51                    description,
52                    description_html,
53                    full,
54                },
55            )
56        })
57        .collect();
58
59    let json = serde_json::to_string(&crate_metadatas).unwrap();
60    let docs_dir = PathBuf::from(&project_root).join("docs");
61    fs::create_dir_all(&docs_dir).unwrap();
62    fs::write(
63        docs_dir.join("dependencies.js"),
64        format!("dependencies = {json}"),
65    )
66    .unwrap();
67}

69/// crate ルートの `//!` doc comment を、要約行（1行目）と本文（2行目以降を HTML レンダリングしたもの）に分ける。
70fn parse_crate_docs(lib_rs_content: &str) -> (Option<String>, Option<String>, Option<String>) {
71    let lines: Vec<&str> = lib_rs_content
72        .lines()
73        .take_while(|line| line.starts_with("//!"))
74        .map(|line| {
75            let rest = &line["//!".len()..];
76            rest.strip_prefix(' ').unwrap_or(rest)
77        })
78        .collect();
79
80    let mut idx = 0;
81    while idx < lines.len() && lines[idx].trim().is_empty() {
82        idx += 1;
83    }
84    if idx >= lines.len() {
85        return (None, None, None);
86    }
87    let description = lines[idx].trim().to_owned();
88    let description_html = render_markdown(&description, true);
89    idx += 1;
90    while idx < lines.len() && lines[idx].trim().is_empty() {
91        idx += 1;
92    }
93    if idx >= lines.len() {
94        return (Some(description), Some(description_html), None);
95    }
96
97    let markdown = lines[idx..].join("\n");
98    let full = render_markdown(&markdown, false);
99    (Some(description), Some(description_html), Some(full))
100}

102const MARKDOWN_OPTIONS: pulldown_cmark::Options = pulldown_cmark::Options::ENABLE_TABLES;
103
104/// Markdown を HTML にレンダリングする。`$...$` / `$$...$$` の数式は、CommonMark のバックスラッシュ
105/// エスケープ規則（`\{` 等の記号直前のバックスラッシュを除去してしまう）で LaTeX が壊れないよう、
106/// パース前に退避させてパース後に生の内容のまま復元する。
107fn render_markdown(markdown: &str, inline_only: bool) -> String {
108    let (protected, math_spans) = protect_math(markdown);
109    let mut html = String::new();
110    pulldown_cmark::html::push_html(
111        &mut html,
112        pulldown_cmark::Parser::new_ext(&protected, MARKDOWN_OPTIONS),
113    );
114    for (i, span) in math_spans.iter().enumerate() {
115        html = html.replace(&math_placeholder(i), &escape_html_minimal(span));
116    }
117    if inline_only {
118        html = strip_outer_block_tag(html.trim());
119    }
120    html
121}
122
123/// 要約行（1行目）は常にインライン表示したいが、`# 見出し`のようにMarkdownの
124/// ブロック要素（`<h1>`〜`<h6>`, `<p>`, `<blockquote>`等）としてレンダリングされることがある。
125/// それらのタグをそのまま埋め込むと表示先の文脈（`<p class="summary">`等）でレイアウトが壊れるため、
126/// 最外周のブロックタグ1つだけを剥がしてテキストとして扱う。
127fn strip_outer_block_tag(html: &str) -> String {
128    if let Some(rest) = html.strip_prefix('<')
129        && let Some(tag_end) = rest.find('>')
130    {
131        let tag_name = rest[..tag_end].split_whitespace().next().unwrap_or("");
132        let closing = format!("</{tag_name}>");
133        if !tag_name.is_empty() && !tag_name.starts_with('/') && html.ends_with(&closing) {
134            return html[tag_end + 2..html.len() - closing.len()].to_owned();
135        }
136    }
137    html.to_owned()
138}

140fn math_placeholder(index: usize) -> String {
141    format!("KATEXMATHPLACEHOLDER{index}")
142}

144/// `$...$`（インライン）・`$$...$$`（ディスプレイ）の数式部分をプレースホルダーに置き換える。
145/// `\$` はエスケープされたただのドル記号として扱い、数式の区切りとは見なさない。
146fn protect_math(markdown: &str) -> (String, Vec<String>) {
147    let chars: Vec<char> = markdown.chars().collect();
148    let mut protected = String::with_capacity(markdown.len());
149    let mut math_spans = Vec::new();
150    let mut i = 0;
151    while i < chars.len() {
152        if chars[i] == '\\' && i + 1 < chars.len() {
153            protected.push(chars[i]);
154            protected.push(chars[i + 1]);
155            i += 2;
156            continue;
157        }
158        if chars[i] == '$' {
159            let display = chars.get(i + 1) == Some(&'$');
160            let delim_len = if display { 2 } else { 1 };
161            let mut j = i + delim_len;
162            let mut end = None;
163            while j < chars.len() {
164                if chars[j] == '\\' && j + 1 < chars.len() {
165                    j += 2;
166                    continue;
167                }
168                if !display && chars[j] == '\n' {
169                    break;
170                }
171                if chars[j] == '$' && (!display || chars.get(j + 1) == Some(&'$')) {
172                    end = Some(j + delim_len);
173                    break;
174                }
175                j += 1;
176            }
177            if let Some(end) = end {
178                let span: String = chars[i..end].iter().collect();
179                protected.push_str(&math_placeholder(math_spans.len()));
180                math_spans.push(span);
181                i = end;
182                continue;
183            }
184        }
185        protected.push(chars[i]);
186        i += 1;
187    }
188    (protected, math_spans)
189}

191fn escape_html_minimal(s: &str) -> String {
192    s.replace('&', "&amp;")
193        .replace('<', "&lt;")
194        .replace('>', "&gt;")
195}

197#[cfg(test)]
198mod tests {

201    #[test]
202    fn tables_are_rendered() {
203        let markdown = "| a | b |\n|---|---|\n| 1 | 2 |\n";
204        let html = render_markdown(markdown, false);
205        assert!(html.contains("<table>"), "got: {html}");
206    }
207
208    #[test]
209    fn latex_punctuation_escapes_survive_markdown() {
210        // `\{`, `\}`, `\,`, `\#` はいずれも CommonMark 的には「記号の前のバックスラッシュ除去」
211        // の対象だが、数式中では KaTeX コマンドとしてバックスラッシュを残す必要がある。
212        let markdown = r"$\# \{\, x \,\}$";
213        let html = render_markdown(markdown, false);
214        assert!(html.contains(r"$\# \{\, x \,\}$"), "got: {html}");
215    }
216
217    #[test]
218    fn angle_brackets_in_math_are_html_escaped() {
219        let markdown = "$a < b$";
220        let html = render_markdown(markdown, false);
221        assert!(html.contains("$a &lt; b$"), "got: {html}");
222    }
223
224    #[test]
225    fn heading_as_description_is_unwrapped_to_plain_text() {
226        let (_, description_html, _) = parse_crate_docs("//! # Manacher's algorithm\n");
227        assert_eq!(description_html.unwrap(), "Manacher's algorithm");
228    }
229
230    #[test]
231    fn description_is_rendered_as_inline_html_without_p_wrapper() {
232        let (_, description_html, _) =
233            parse_crate_docs("//! [`Vec<u64>`] の話 $O(n)$\n//!\n//! 本文。\n");
234        let html = description_html.unwrap();
235        assert!(!html.contains("<p>"), "got: {html}");
236        assert!(html.contains("<code>Vec&lt;u64&gt;</code>"), "got: {html}");
237        assert!(html.contains("$O(n)$"), "got: {html}");
238    }
