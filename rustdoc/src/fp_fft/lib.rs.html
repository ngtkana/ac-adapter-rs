1//! 有限体 $𝔽_P$ 上の高速フーリエ変換（FFT）。
2//!
3//! Cooley-Tukey アルゴリズムで、多項式乗算を $O(n^2)$ から $O(n \log n)$ に削減。
4//!
5//! # 仕様
6//!
7//! 配列 $(x_0, \ldots, x_{n-1})$ に対して：
8//! - `fft`: $X_i = \sum_{j=0}^{n-1} x_j \cdot w^{ij}$（$w$ は $n$ 次原始単位根）
9//! - `ifft`: `fft` の逆変換（$1/n$ でスケール）
10//!
11//! # 例
12//!
13//! ```
14//! use fp::fp_new;
15//! use fp_fft::fft;
16//!
17//! const P: u64 = 998_244_353;
18//! let mut data = [fp_new::<P>(1), fp_new::<P>(2)];
19//! fft(&mut data);
20//! assert_eq!(data[0], fp_new::<P>(3)); // 1 + 2
21//! assert_eq!(data[1], fp_new::<P>(998244352)); // 1 - 2 ≡ -1 (mod P)
22//! ```
23//!
24//! # 公開 API
25//!
26//! - [`fft`]: 前方フーリエ変換、$O(n \log n)$
27//! - [`ifft`]: 逆フーリエ変換、$O(n \log n)$
28

34const DIADIC_ROOTS_BUFFER_LEN: usize = 64;
35
36const fn find_primitive_root<const P: u64>() -> Fp<P> {
37    let mut x = fp_new(2);
38    while x.value() != P {
39        if x.pow((P - 1) / 2).value() != 1 {
40            return x;
41        }
42        x.add_assign(fp_new(1));
43    }
44    panic!("primitive root not found");
45}
46
47const fn build_diadic_roots<const P: u64>(root: Fp<P>) -> [Fp<P>; DIADIC_ROOTS_BUFFER_LEN] {
48    let mut result = [fp_new(0); DIADIC_ROOTS_BUFFER_LEN];
49    let k = (P - 1).trailing_zeros();
50    let mut i = k as usize;
51    result[i] = root.pow((P - 1) >> k);
52    while i != 0 {
53        result[i - 1] = result[i].mul(result[i]);
54        i -= 1;
55    }
56    result
57}
58
59trait DiadicRootsTrait<const P: u64> {
60    const VALUE: [Fp<P>; DIADIC_ROOTS_BUFFER_LEN];
61}
62enum DiadicRoots<const P: u64> {}
63impl<const P: u64> DiadicRootsTrait<P> for DiadicRoots<P> {
64    const VALUE: [Fp<P>; DIADIC_ROOTS_BUFFER_LEN] = build_diadic_roots(find_primitive_root());
65}

67/// FFT をします。周波数間引き(Sande–Tukey)で、出力はbit-reversedです。
68///
69/// 内部で [`build_twiddle_factors`] と [`fft_with_twiddle_factors`] が呼ばれます。
70///
71/// # Examples
72///
73/// ```
74/// use fp::fp_new;
75/// use fp_fft::fft;
76///
77/// const P: u64 = 998_244_353;
78///
79/// let mut a = [fp_new::<P>(3), fp_new::<P>(5)];
80/// fft(&mut a);
81/// assert_eq!(a[0], fp_new::<P>(8)); // 3 + 5
82/// assert_eq!(a[1], fp_new::<P>(998244351)); // 3 - 5 ≡ -1
83/// ```
84pub fn fft<const P: u64>(items: &mut [Fp<P>]) {
85    let twiddle_factors = build_twiddle_factors(items.len());
86    fft_with_twiddle_factors(items, &twiddle_factors);
87}

89/// Twiddle factor 前計算済みの場合の、[`fft`]。
90///
91/// # Examples
92///
93/// ```
94/// use fp::fp_new;
95/// use fp_fft::build_twiddle_factors;
96/// use fp_fft::fft_with_twiddle_factors;
97///
98/// const P: u64 = 998_244_353;
99///
100/// let mut a = [fp_new::<P>(3), fp_new::<P>(5)];
101/// let twiddle_factors = build_twiddle_factors(2);
102/// fft_with_twiddle_factors(&mut a, &twiddle_factors);
103///
104/// assert_eq!(a[0], fp_new::<P>(8)); // 3 + 5
105/// assert_eq!(a[1], fp_new::<P>(998244351)); // 3 - 5 ≡ -1
106/// ```
107pub fn fft_with_twiddle_factors<const P: u64>(items: &mut [Fp<P>], twiddle_factors: &[Fp<P>]) {
108    assert!(items.len().is_power_of_two());
109    assert!(items.len().trailing_zeros() <= (P - 1).trailing_zeros());
110    for n in successors(Some(items.len()), |&n| Some(n / 2)).take_while(|&n| n >= 2) {
111        for chunk in items.chunks_mut(n) {
112            for i in 0..n / 2 {
113                let [a, b] = unsafe { chunk.get_disjoint_unchecked_mut([i, i + n / 2]) };
114                [*a, *b] = [*a + *b, *a - *b];
115                *b *= twiddle_factors[n + i];
116            }
117        }
118    }
119}

121/// IFFT をします。時間間引き(Cooley–Tukey)で、入力はbit-reversed想定です。
122///
123/// 内部で [`build_twiddle_factors`] と [`fft_with_twiddle_factors`] が呼ばれます。
124///
125/// # Examples
126///
127/// ```
128/// use fp::fp_new;
129/// use fp_fft::ifft;
130///
131/// const P: u64 = 998_244_353;
132///
133/// let mut a = [fp_new::<P>(12), fp_new::<P>(4)];
134/// ifft(&mut a);
135///
136/// assert_eq!(a[0], fp_new::<P>(8));
137/// assert_eq!(a[1], fp_new::<P>(4));
138/// ```
139pub fn ifft<const P: u64>(items: &mut [Fp<P>]) {
140    let twiddle_factors = build_twiddle_factors(items.len());
141    ifft_with_twiddle_factors(items, &twiddle_factors);
142}

144/// Twiddle factor 前計算済みの場合の、[`ifft`]。
145///
146/// # Examples
147///
148/// ```
149/// use fp::fp_new;
150/// use fp_fft::build_twiddle_factors;
151/// use fp_fft::ifft_with_twiddle_factors;
152///
153/// const P: u64 = 998_244_353;
154///
155/// let mut a = [fp_new::<P>(12), fp_new::<P>(4)];
156/// let twiddle_factors = build_twiddle_factors(2);
157/// ifft_with_twiddle_factors(&mut a, &twiddle_factors);
158///
159/// assert_eq!(a[0], fp_new::<P>(8));
160/// assert_eq!(a[1], fp_new::<P>(4));
161/// ```
162pub fn ifft_with_twiddle_factors<const P: u64>(items: &mut [Fp<P>], twiddle_factors: &[Fp<P>]) {
163    let items_len = items.len();
164    assert!(items_len.is_power_of_two());
165    assert!(items_len.trailing_zeros() <= (P - 1).trailing_zeros());
166    for n in successors(Some(2), |&n| Some(2 * n)).take_while(|&n| n <= items_len) {
167        for chunk in items.chunks_mut(n) {
168            for i in 0..n / 2 {
169                let [a, b] = unsafe { chunk.get_disjoint_unchecked_mut([i, i + n / 2]) };
170                *b *= twiddle_factors[2 * n - i];
171                [*a, *b] = [*a + *b, *a - *b];
172            }
173        }
174    }
175    let len_inv = fp_new(items.len() as u64).inv();
176    for item in items {
177        *item *= len_inv;
178    }
179}

181/// Twiddle factors を計算する(FFT用)
182///
183/// 長さ $2n + 1$ の配列ができます。最初の $1$ つは使わない場所。最後の $1$ つは番兵です。
184///
185/// $$
186/// t _ { 2 ^ p + i } = e( i / 2 ^ p)
187/// $$
188///
189/// ただし $e(a / b)$ は $1$ の原始 $b$ 乗根の $a$ 乗です。
190///
191/// # Examples
192///
193/// ```
194/// use fp::fp_new;
195/// use fp_fft::build_twiddle_factors;
196/// use fp_fft::fft_with_twiddle_factors;
197///
198/// const P: u64 = 998_244_353;
199///
200/// let mut a = [fp_new::<P>(3), fp_new::<P>(5)];
201/// let twiddle_factors = build_twiddle_factors(2);
202/// fft_with_twiddle_factors(&mut a, &twiddle_factors);
203///
204/// assert_eq!(a[0], fp_new::<P>(8)); // 3 + 5
205/// assert_eq!(a[1], fp_new::<P>(998244351)); // 3 - 5 ≡ -1
206/// ```
207pub fn build_twiddle_factors<const P: u64>(n: usize) -> Vec<Fp<P>> {
208    let mut twiddle_factors = vec![fp_new::<P>(1); 2 * n + 1];
209    for n in successors(Some(2), |&x| Some(2 * x)).take_while(|&x| x <= n) {
210        let w = DiadicRoots::VALUE[n.trailing_zeros() as usize];
211        for i in 0..n / 2 {
212            twiddle_factors[n + i * 2] = twiddle_factors[n / 2 + i];
213            twiddle_factors[n + i * 2 + 1] = twiddle_factors[n + i * 2] * w;
214        }
215    }
216    twiddle_factors
217}
