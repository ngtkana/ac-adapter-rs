1//! 変数の値を見出し付きで標準エラー出力に出す、競プロ用デバッグユーティリティ。
2//!
3//! `lg!` は式をそのまま `名前 = 値` の形に整形するだけだが、`table!`/`vmap!`/`hmap!`/`hvec!`/`vvec!` は
4//! 2 次元配列やマップを罫線付きの表に組んで出す。見出しには `stringify!` で得たソースコード上の式の文字列を使い、
5//! 値は `format` を通す。`format` は `u32::MAX` や `i64::MIN` のような番兵値を `*` に、`bool` を `#`/`.` に
6//! 変換するため、「未到達を表す番兵値」や「訪問済みフラグ」の配列を目視でそのまま読める。
7//!
8//! # 仕様
9//!
10//! - `lg!`: 複数の式を `行番号 ❯ 式1 = 値1, 式2 = 値2` の形式で 1 行出力
11//! - `table!`: 2 次元配列を罫線付きの表として出力
12//! - `vmap!` / `hmap!`: マップを縦 / 横の表として出力
13//! - `vvec!` / `hvec!`: 複数の `Vec` を並べて縦 / 横の表として出力（`@field` で構造体のフィールドを抽出可能）
14//! - `format`: `Debug` 文字列を番兵値・`bool`・`Option`・引用符について整形
15//! - `bools`: `bool` の列を `[.#..]` のようなビット列表記に変換
16//!
17//! # 例
18//!
19//! ```
20//! use lg::lg;
21//! let a = 42;
22//! let v = vec![1, 2, 3];
23//! lg!(a, v);
24//! ```

40/// `bool` の列を `[.#..]` の形式に変換する。真を `#`、偽を `.` で表す。
41///
42/// # 例
43///
44/// ```
45/// use lg::bools;
46/// assert_eq!(bools([true, false, true]), "[#.#]");
47/// ```
48pub fn bools<B, I>(iter: I) -> String
49where
50    B: Borrow<bool>,
51    I: IntoIterator<Item = B>,
52{
53    format!(
54        "[{}]",
55        iter.into_iter()
56            .map(|b| ['.', '#'][usize::from(*(b.borrow()))])
57            .collect::<String>(),
58    )
59}

61/// 表のセルの寄せ方向を、文字列が数値として解釈できるかで決める。
62///
63/// 浮動小数点として parse できれば右寄せ、できなければ左寄せにする。数値の列を右揃えで見やすくするための判定。
64pub fn align_of(s: &str) -> Align {
65    // To improve this: https://doc.rust-lang.org/reference/tokens.html#floating-point-literals
66    match s.parse::<f64>() {
67        Ok(_) => Align::Right,
68        Err(_) => Align::Left,
69    }
70}

72/// 複数の式を評価し、`行番号 ❯ 式1 = 値1, 式2 = 値2` の形式で標準エラー出力に 1 行出す。
73///
74/// 各値は `format` で整形する。競プロのデバッグで、複数の変数を毎回 `eprintln!` を書かずに確認する用途に使う。
75///
76/// # 例
77///
78/// ```
79/// use lg::lg;
80/// let a = 1;
81/// let b = vec![2, 3];
82/// lg!(a, b); // 標準エラーに `<行番号> ❯ a = 1, b = [2, 3]` を出力
83/// ```
84#[macro_export]
85macro_rules! lg {
86    (@contents $head:expr $(, $tail:expr)*) => {{
87        $crate::__lg_internal!($head);
88        $(
89            eprint!(",");
90            $crate::__lg_internal!($tail);
91        )*
92        eprintln!();
93    }};
94    ($($expr:expr),* $(,)?) => {{
95        eprint!("{} \u{276f}", line!());
96        $crate::lg!(@contents $($expr),*)
97    }};
98}

100/// `lg!` の内部実装。1 個の式を `式 = 値` の形式で出力する。外部から直接使わない。
101#[doc(hidden)]
102#[macro_export]
103macro_rules! __lg_internal {
104    ($value:expr) => {{
105        match $value {
106            head => {
107                eprint!(" {} = {}", stringify!($value), $crate::format(&head));
108            }
109        }
110    }};
111}

113/// 2 次元配列（`Vec<Vec<T>>` など、行の集まりを 2 重にイテレートできる型）を罫線付きの表として標準エラー出力に出す。
114///
115/// 見出しには式のソース文字列（先頭の `&` は除く）を使い、各セルの値は `format` で整形する。
116///
117/// # 例
118///
119/// ```
120/// use lg::table;
121/// let v = vec![vec![1, 2], vec![3, 4]];

125macro_rules! table {
126    ($vec2:expr) => {
127        eprint!(
128            "{}",
129            $crate::vec2($crate::remove_ampersand(stringify!($vec2)), $vec2)
130        );
131    };

172/// 複数の `Vec`（またはイテレート可能な列）を、それぞれ 1 列とする表にして標準エラー出力に出す。列見出しには式のソース文字列を、行見出しにはインデックスを使う。
173///
174/// `@field name expr` を前置すると、要素が構造体のときに指定フィールドだけを抽出して列にできる。
175///
176/// # 例
177///
178/// ```
179/// use lg::vvec;
180/// let a = [1, 2, 3];
181/// let b = [4, 5, 6];
182/// vvec!(&a, &b);
183/// ```
184#[macro_export]
185macro_rules! vvec {
186    ($($(@field $field:ident)* $vecs:expr),+ $(,)?) => {
187        let mut vecs = Vec::new();
188        $(
189            let name = $crate::remove_ampersand(stringify!($vecs));
190            #[allow(unused_mut, unused_assignments)]
191            let mut has_field = false;
192            $(
193                #[allow(unused_mut, unused_assignments)]
194                {
195                    let mut name = name.to_owned();
196                    has_field = true;
197                    name.push_str(".");
198                    name.push_str(stringify!($field));
199                    let values = (&$vecs).into_iter().map(|v| $crate::format(&v.$field)).collect::<Vec<_>>();
200                    vecs.push((name, values))
201                }
202            )*
203            if !has_field {
204                let values = (&$vecs).into_iter().map(|v| $crate::format(&v)).collect::<Vec<_>>();
205                vecs.push((name.to_owned(), values))
206            }
207        )+
208        eprint!("{}", $crate::vvec(&vecs));
209    };
210}

252/// 文字列先頭の `&` をすべて取り除く。
253///
254/// マクロが `stringify!(&&x)` のような参照付きの式を見出しに使う際、表示を `x` に揃えるために使う。
255///
256/// # 例
257///
258/// ```
259/// use lg::remove_ampersand;
260/// assert_eq!(remove_ampersand("&&x"), "x");
261/// ```
262pub fn remove_ampersand(mut s: &str) -> &str {
263    while let Some(t) = s.strip_prefix('&') {
264        s = t;
265    }
266    s
267}

269/// 値を `Debug` で文字列化し、読みやすく整形する。
270///
271/// 次の変換を順に適用する。
272///
273/// - `u32::MAX`, `u64::MAX`, `u128::MAX`, `i32::MIN`/`MAX`, `i64::MIN`/`MAX`, `i128::MAX` を番兵値とみなし `*` に置換
274/// - `None` を `*` に、`Some(x)` を `x` に置換（入れ子も再帰的に剥がす）
275/// - `true`/`false` を `#`/`.` に置換
276/// - 文字列を囲む `"` を除去
277///
278/// # 例
279///
280/// ```
281/// use lg::format;
282/// assert_eq!(format(&Some(3)), "3");
283/// assert_eq!(format(&u32::MAX), "*");
284/// assert_eq!(format(&true), "#");
285/// ```
286pub fn format<T: fmt::Debug>(t: &T) -> String {
287    let s = format!("{t:?}")
288        .replace("340282366920938463463374607431768211455", "*") // u128
289        .replace("170141183460469231731687303715884105727", "*") // i128
290        .replace("18446744073709551615", "*") // u64
291        .replace("9223372036854775807", "*") // i64
292        .replace("-9223372036854775808", "*") // i64
293        .replace("4294967295", "*") // u32
294        .replace("2147483647", "*") // i32
295        .replace("-2147483648", "*") // i32
296        .replace("None", "*")
297        .replace("true", "#")
298        .replace("false", ".");
299    let mut s = s.as_str();
300    while s.starts_with("Some(") {
301        s = s.strip_prefix("Some(").unwrap();
302        s = s.strip_suffix(')').unwrap();
303    }
304    while s.len() > 2 && s.starts_with('"') && s.ends_with('"') {
305        s = s.strip_prefix('"').unwrap();
306        s = s.strip_suffix('"').unwrap();
307    }
308    s.to_owned()
309}
