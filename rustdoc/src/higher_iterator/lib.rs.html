1//! 多次元コンテナを固定長インデックス配列付きで列挙するイテレータ。
2
3/// 多次元コンテナの各要素を `([usize; D], Item)` として列挙する。
4///
5/// `D` は次元数。インデックスは `Vec` でなく固定長配列 `[usize; D]` で返す。
6/// `Self` は `Vec` に限定されず、`&'a Self` がネストした `IntoIterator` になっている型であれば実装される。
7///
8/// 内側の型自体が `Copy`（固定長配列など）の場合、`D` の異なる複数の実装が同時に候補となり、
9/// 型推論だけでは `D` が決まらないことがある。その場合は
10/// `HigherIterator::<D>::higher_enumerate(&x)` のようにフルパスで `D` を指定する。
11///
12/// `D` は 1〜8 の実装を用意している。さらに大きい `D` が必要な場合は
13/// `impl_higher_iterators!(i0, ..., i8)` の ident 列を1つ増やすだけでよい。
14///
15/// # Examples
16///
17/// 1 次元（`Vec<i32>`）:
18///
19/// ```
20/// use higher_iterator::HigherIterator;
21///
22/// let v = vec![10, 20, 30];
23/// let result = v.higher_enumerate().collect::<Vec<_>>();
24/// assert_eq!(result, vec![([0], 10), ([1], 20), ([2], 30)]);
25/// ```
26///
27/// 2 次元（`Vec<Vec<i32>>`）:
28///
29/// ```
30/// use higher_iterator::HigherIterator;
31///
32/// let v = vec![vec![1, 2], vec![3, 4, 5]];
33/// let result = v.higher_enumerate().collect::<Vec<_>>();
34/// assert_eq!(result, vec![
35///     ([0, 0], 1), ([0, 1], 2),
36///     ([1, 0], 3), ([1, 1], 4), ([1, 2], 5),
37/// ]);
38/// ```
39///
40/// 3 次元（`Vec<Vec<Vec<i32>>>`）:
41///
42/// ```
43/// use higher_iterator::HigherIterator;
44///
45/// let v = vec![vec![vec![1, 2], vec![3]], vec![vec![4, 5, 6]]];
46/// let result = v.higher_enumerate().collect::<Vec<_>>();
47/// assert_eq!(result, vec![
48///     ([0, 0, 0], 1), ([0, 0, 1], 2),
49///     ([0, 1, 0], 3),
50///     ([1, 0, 0], 4), ([1, 0, 1], 5), ([1, 0, 2], 6),
51/// ]);
52/// ```
53pub trait HigherIterator<'a, const D: usize> {
54    /// 列挙される要素の型。
55    type Item: Copy + 'a;
56
57    /// 各要素を `([usize; D], Item)` として列挙するイテレータを返す。
58    fn higher_enumerate(&'a self) -> impl Iterator<Item = ([usize; D], Self::Item)>;
59}
60
61/// `higher_enumerate` の本体を、ネストした `enumerate().flat_map(...)` の連鎖として組み立てる。
62///
63/// 末端では `enumerate().map(...)` でインデックス配列 `[$acc.., $last]` と要素を返す。
64/// 途中段では `enumerate().flat_map(...)` で自分のインデックスを `move` キャプチャしつつ、
65/// 内側のコンテナに対して同じ処理を再帰的に展開する。
66macro_rules! higher_iterator_body {
67    ($self_expr:expr; [$($acc:ident),*]; $last:ident) => {
68        $self_expr
69            .into_iter()
70            .enumerate()
71            .map(move |($last, &x)| ([$($acc,)* $last], x))
72    };
73    ($self_expr:expr; [$($acc:ident),*]; $head:ident, $($tail:ident),+) => {
74        $self_expr.into_iter().enumerate().flat_map(move |($head, row)| {
75            higher_iterator_body!(row; [$($acc,)* $head]; $($tail),+)
76        })
77    };
78}

80/// `D` 次元分の `HigherIterator` 実装を生成する。
81///
82/// `where` 句はマクロ内で再帰的に、`&'a I: IntoIterator` を1段ずつネストした
83/// 境界として組み立てる（例えば `D = 3` なら `&'a I`, `<&'a I as IntoIterator>::Item`,
84/// `<<&'a I as IntoIterator>::Item as IntoIterator>::Item` の3段）。
85/// 最後の段だけ `IntoIterator<Item = &'a T>` を要求する。
86///
87/// `$D` は `usize` の式（[`impl_higher_iterators`] からは `1 + 1 + ...` の形で渡される）。
88/// トレイトの const generic 引数は式なら `{ }` で囲む必要があるが、配列長は不要。
89/// `$D` が単なるリテラルに簡約される場合（`D = 1`）は `{ }` が不要になり
90/// `unused_braces` の warning が出るため、生成する impl に `#[allow(unused_braces)]` を付ける。
91macro_rules! impl_higher_iterator {
92    ($D:expr; $($idx:ident),+ $(,)?) => {
93        impl_higher_iterator!(@emit $D; ($($idx),+); &'a I; []; $($idx),+);
94    };
95    (@emit $D:expr; ($($all:ident),+); $cur:ty; [$($bounds:tt)*]; $head:ident, $($tail:ident),+) => {
96        impl_higher_iterator!(
97            @emit $D; ($($all),+); <$cur as IntoIterator>::Item;
98            [$($bounds)* $cur: IntoIterator,];
99            $($tail),+
100        );
101    };
102    (@emit $D:expr; ($($all:ident),+); $cur:ty; [$($bounds:tt)*]; $last:ident) => {
103        #[allow(unused_braces)]
104        impl<'a, I: ?Sized + 'a, T: Copy + 'a> HigherIterator<'a, { $D }> for I
105        where
106            $($bounds)*
107            $cur: IntoIterator<Item = &'a T>,
108        {
109            type Item = T;
110
111            fn higher_enumerate(&'a self) -> impl Iterator<Item = ([usize; $D], Self::Item)> {
112                higher_iterator_body!(self; []; $($all),+)
113            }
114        }
115    };
116}

118/// [`impl_higher_iterator`] を `D = 1` から idents の個数分だけ再帰的に呼び出す。
119///
120/// idents 列を1つずつ先頭から取り出し、その時点までの累積個数（`1`, `1 + 1`, ...）を
121/// `D` として渡す。呼び出しコードは idents 列を1度書くだけでよく、`D` を明示する必要がない。
122macro_rules! impl_higher_iterators {
123    (@step $count:expr; [$($acc:ident),+]) => {
124        impl_higher_iterator!($count; $($acc),+);
125    };
126    (@step $count:expr; [$($acc:ident),+] $head:ident $(, $tail:ident)*) => {
127        impl_higher_iterator!($count; $($acc),+);
128        impl_higher_iterators!(@step $count + 1; [$($acc,)+ $head] $($tail),*);
129    };
130    ($head:ident $(, $tail:ident)*) => {
131        impl_higher_iterators!(@step 1; [$head] $($tail),*);
132    };
133}
134
135impl_higher_iterators!(i0, i1, i2, i3, i4, i5, i6, i7);

253    fn test_4d_shape_2x2x2x2() {
254        // 各要素の値を i0*8 + i1*4 + i2*2 + i3 で置いた 2x2x2x2 の入れ子 Vec。
255        let v: Vec<Vec<Vec<Vec<i32>>>> = (0..2)
256            .map(|i0| {
257                (0..2)
258                    .map(|i1| {
259                        (0..2)
260                            .map(|i2| (0..2).map(|i3| i0 * 8 + i1 * 4 + i2 * 2 + i3).collect())
261                            .collect()
262                    })
263                    .collect()
264            })
265            .collect();
266        let result = v.higher_enumerate().collect::<Vec<_>>();
267        assert_eq!(result.len(), 16);
268        for (idx, x) in result {
269            let [i0, i1, i2, i3] = idx.map(|i| i as i32);
270            assert_eq!(x, i0 * 8 + i1 * 4 + i2 * 2 + i3);
271        }
272    }
