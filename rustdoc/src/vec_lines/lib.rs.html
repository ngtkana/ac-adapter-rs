1//! 傾き単調な直線群を管理し、任意の $x$ における最大値・最小値クエリに答える Convex Hull Trick。
2//!
3//! 直線を追加するたびに、傾きの単調性と定数項の優劣から不要な直線をスタック的に取り除き、
4//! 常に「下側（または上側）包絡線を構成する直線だけ」を `Vec` で保持する。クエリは黄金分割探索で
5//! 答える。クエリの $x$ が挿入順に単調に動く場合は [`VecLines::get`] としゃくとり法を組み合わせる
6//! ことで、`eval_gcc` よりさらに高速に処理できる。
7//!
8//! # 仕様
9//!
10//! [`Constraint`] で傾きの単調方向を指定する（[`VecLinesDecreasing`] は単調減少、
11//! [`VecLinesIncreasing`] は単調増加）。
12//!
13//! - `push` は直線 $y = ax + b$ を `[a, b]` で追加する。傾きの単調性に反すると panic する
14//! - `eval_gcc` は $x$ における最適値（[`VecLinesDecreasing`] は最小値、
15//!   [`VecLinesIncreasing`] は最大値）を返す
16//! - `get` は `index` 番目の直線 [`Line`] を返す
17//! - `len`、`is_empty`、`iter_copied` で内部状態を参照する
18//!
19//! 傾きが等しい直線を追加した場合、定数項が真に改善していれば直前の直線を置き換えてから
20//! 通常の不要直線除去を行い、改善していなければ追加を無視する。
21//!
22//! # 例
23//!
24//! ```
25//! use vec_lines::VecLinesDecreasing;
26//!
27//! let mut lines = VecLinesDecreasing::<i32>::new();
28//! lines.push([1, 0]); // y = x
29//! lines.push([0, 10]); // y = 10
30//! lines.push([-1, 30]); // y = -x + 30
31//!
32//! assert_eq!(lines.eval_gcc(-10), Some(-10)); // x = -10 での最小値
33//! assert_eq!(lines.eval_gcc(15), Some(10)); // y = 10 での最小値
34//! assert_eq!(lines.eval_gcc(40), Some(-10)); // -x + 30 での最小値
35//! ```
36//!
37//! # 計算量
38//!
39//! - `push`: 償却 $O(1)$
40//! - `eval_gcc`: $O(\log n)$（$n$ は管理している直線の本数）

50pub type VecLinesDecreasing<T> = VecLines<T, DecreasingTilt>;

54/// 傾き単調な直線群。詳細はクレートの説明を参照。
55#[derive(Clone, Debug, Hash, PartialEq, Eq)]
56pub struct VecLines<T, C> {
57    lines: Vec<Line<T>>,
58    _marker: PhantomData<fn(C) -> C>,
59}
60impl<T: Signed, C: Constraint> VecLines<T, C> {
61    /// 空の直線群を構築する。
62    ///
63    /// # 例
64    /// ```
65    /// use vec_lines::VecLinesDecreasing;
66    /// let lines = VecLinesDecreasing::<i32>::new();
67    /// assert!(lines.is_empty());
68    /// ```
69    pub fn new() -> Self {
70        Self {
71            lines: Vec::new(),
72            _marker: PhantomData,
73        }
74    }
75
76    /// 直線を 1 本も持たなければ `true` を返す。
77    pub fn is_empty(&self) -> bool {
78        self.lines.is_empty()
79    }
80
81    /// 管理している直線の本数を返す。不要な直線は自動的に削除されるため、
82    /// [`push`](Self::push) のたびに単調増加するとは限らない。
83    pub fn len(&self) -> usize {
84        self.lines.len()
85    }
86
87    /// `index` 番目の直線 [`Line`] を返す。範囲外なら `None`。
88    ///
89    /// # 例
90    /// ```
91    /// use vec_lines::VecLinesDecreasing;
92    /// let mut lines = VecLinesDecreasing::<i32>::new();
93    /// lines.push([1, 0]);
94    /// assert_eq!(lines.get(0).unwrap().eval(100), 100);
95    /// assert!(lines.get(1).is_none());
96    /// ```
97    pub fn get(&self, index: usize) -> Option<Line<T>> {
98        self.lines.get(index).copied()
99    }
100
101    /// 直線 $y = ax + b$ を `[a, b]` で追加する。
102    ///
103    /// # Panics
104    ///
105    /// マーカー `C` の定める傾きの単調性に違反するとき。
106    ///
107    /// # 計算量
108    ///
109    /// 償却 $O(1)$。
110    pub fn push(&mut self, line: [T; 2]) {
111        assert!(
112            self.lines.last().is_none_or(|prv| C::ok(*prv, Line(line))),
113            "傾きの単調性に違反しています。"
114        );
115        if let Some(&Line(prv)) = self.lines.last()
116            && prv[0] == line[0]
117        {
118            if C::strictly_better(line[1], prv[1]) {
119                self.lines.pop();
120            } else {
121                return;
122            }
123        }
124        self.lines.push(Line(line));
125        while 3 <= self.lines.len()
126            && weakly_convex(
127                *<&[Line<_>; 3]>::try_from(&self.lines[self.lines.len() - 3..]).unwrap(),
128            )
129        {
130            let p = self.lines.pop().unwrap();
131            self.lines.pop().unwrap();
132            self.lines.push(p);
133        }
134    }
135
136    /// 黄金分割探索で $x$ における最適値を計算する。直線が 1 本もなければ `None`。
137    ///
138    /// # 計算量
139    ///
140    /// $O(\log n)$（$n$ は管理している直線の本数）。
141    pub fn eval_gcc(&self, x: T) -> Option<T> {
142        if self.lines.is_empty() {
143            return None;
144        }
145        let mut i0 = -1;
146        let mut i3 = self.len() as isize;
147        while 3 <= i3 - i0 {
148            let [i1, i2] = golden_section([i0, i3]);
149            if C::strictly_better(
150                self.lines[i1 as usize].eval(x),
151                self.lines[i2 as usize].eval(x),
152            ) {
153                i3 = i2;
154            } else {
155                i0 = i1;
156            }
157        }
158        assert!((2..=3).contains(&(i3 - i0)));
159        let y1 = self.lines[(i0 + 1) as usize].eval(x);
160        let y2 = self.lines[(i3 - 1) as usize].eval(x);
161        Some(if C::strictly_better(y1, y2) { y1 } else { y2 })
162    }
163
164    /// 管理している直線を順番に返すイテレータを返す。
165    ///
166    /// # 例
167    /// ```
168    /// use vec_lines::VecLinesDecreasing;
169    /// let mut lines = VecLinesDecreasing::<i32>::new();
170    /// lines.push([1, 0]);
171    /// lines.push([0, 10]);
172    /// let coeffs = lines.iter_copied().map(|l| l.into_coeff()).collect::<Vec<_>>();
173    /// assert_eq!(coeffs, vec![[1, 0], [0, 10]]);
174    /// ```
175    pub fn iter_copied(&self) -> impl '_ + Iterator<Item = Line<T>> {
176        self.lines.iter().copied()
177    }
178}

180/// 一次関数 $ax + b$ を `[a, b]` の形で保持する。中身は `.0` でも [`Line::into_coeff`] でも取れる。
181#[derive(Clone, Debug, Default, Hash, PartialEq, Eq, Copy)]
182pub struct Line<T>(pub [T; 2]);
183impl<T: Signed> Line<T> {
184    /// $x$ における値 $ax + b$ を計算する。
185    ///
186    /// # 例
187    /// ```
188    /// use vec_lines::Line;
189    /// let line = Line([2, 10]);
190    /// assert_eq!(line.eval(2), 14);
191    /// ```
192    pub fn eval(self, x: T) -> T {
193        self.0[0] * x + self.0[1]
194    }
195
196    /// 係数 `[a, b]` を返す。
197    pub fn into_coeff(self) -> [T; 2] {
198        self.0
199    }
200}
201
202fn weakly_convex<T: Signed>(
203    [Line([a0, b0]), Line([a1, b1]), Line([a2, b2])]: [Line<T>; 3],
204) -> bool {
205    (a2 - a1) * (b1 - b0) <= (a1 - a0) * (b2 - b1)
206}
207
208fn golden_section([i0, i3]: [isize; 2]) -> [isize; 2] {
209    let d = ((i3 - i0) as f64 * (5_f64.sqrt() - 1.0) / 2.0).ceil() as isize;
210    [i3 - d, i0 + d]
211}

213/// 傾きがどちら向きに単調かを表すマーカートレイト。
214pub trait Constraint: Clone + Debug + Hash + PartialEq {
215    fn ok<T: Signed>(prv: Line<T>, crr: Line<T>) -> bool;
216    fn strictly_better<T: Signed>(x: T, y: T) -> bool;
217}
218/// 傾き単調減少を意味するマーカー型。
219#[derive(Clone, Debug, Hash, PartialEq, Eq)]
220pub enum DecreasingTilt {}
221/// 傾き単調増加を意味するマーカー型。
222#[derive(Clone, Debug, Hash, PartialEq, Eq)]
223pub enum IncreasingTilt {}
224impl Constraint for DecreasingTilt {
225    fn ok<T: Signed>(Line([a0, _]): Line<T>, Line([a1, _]): Line<T>) -> bool {
226        a0 >= a1
227    }
228
229    fn strictly_better<T: Signed>(x: T, y: T) -> bool {
230        x < y
231    }
232}
233impl Constraint for IncreasingTilt {
234    fn ok<T: Signed>(Line([a0, _]): Line<T>, Line([a1, _]): Line<T>) -> bool {
235        a0 <= a1
236    }
237
238    fn strictly_better<T: Signed>(x: T, y: T) -> bool {
239        x > y
240    }
241}

243/// [`VecLines`] が要素として扱える符号つき整数。標準の符号つき整数型に実装済み。
244pub trait Signed:
245    Debug
246    + Clone
247    + Copy
248    + Default
249    + Hash
250    + PartialOrd
251    + Add<Output = Self>
252    + Sub<Output = Self>
253    + Mul<Output = Self>
254{
255}
256macro_rules! impl_signed {
257    ($($T:ty),* $(,)?) => {$(
258        impl Signed for $T {}
259    )*}
260}
261impl_signed! { i8, i16, i32, i64, i128, isize }
262
263impl<T: Signed, C: Constraint> Default for VecLines<T, C> {
264    fn default() -> Self {
265        Self::new()
266    }

295    #[test]
296    fn test_decreasing_tilt() {
297        let mut rng = StdRng::seed_from_u64(42);
298        for _ in 0..100 {
299            let mut lines = VecLinesDecreasing::new();
300            let mut raw_lines = Vec::new();
301            let n = rng.gen_range(1..20);
302            for x in 0..3 * n {
303                let x = x / 3;
304                let y = rng.gen_range(0..(n * n));
305                let one = -2 * x;
306                let zero = x * x + y;
307                raw_lines.push([one, zero]);
308                lines.push([one, zero]);
309            }
310            let mut i = 0;
311            for x in 0..n {
312                // 全探索
313                let expected = lines.iter_copied().map(|line| line.eval(x)).min().unwrap();
314
315                // しゃくとり法
316                let mut result = lines.get(i).unwrap().eval(x);
317                while let Some(swp) = lines.get(i + 1).map(|line| line.eval(x)) {
318                    if result <= swp {
319                        break;
320                    }
321                    result = swp;
322                    i += 1;
323                }
324                assert_eq!(result, expected);
325
326                // 黄金分割探索
327                let result = lines.eval_gcc(x).unwrap();
328                assert_eq!(result, expected);
329            }
330        }
331    }
