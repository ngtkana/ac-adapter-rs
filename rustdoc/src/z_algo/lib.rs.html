1//! 文字列と各接尾辞との最長共通接頭辞長を線形時間で求める Z algorithm。
2//!
3//! 既知の一致区間（過去に見つけた $[l, r)$ で `s[l..r]` が `s` の接頭辞と一致）を使い回すことで、
4//! 各添字での比較開始位置を後戻りさせずに済ませ、全体の比較回数を $O(n)$ に抑える。
5//! 具体的には、区間内にある添字 $i$ の $z_i$ は「区間内で対応する接頭辞の値」と
6//! 「区間の残り長さ」の小さい方まではコピーで確定でき、一致するかもしれない残りだけを
7//! 実際に文字比較して区間を更新する。
8//!
9//! # 仕様
10//!
11//! [`z_algo`]`(s)` は長さ $n$ の列 `s` に対し、長さ $n$ の Z 配列 `z` を返す。
12//! `z[i]` は `s` と `s[i..]` の最長共通接頭辞の長さ（$z_0 = n$）。
13//!
14//! # 例
15//!
16//! ```
17//! use z_algo::z_algo;
18//! let z = z_algo("abcabca".as_bytes());
19//! assert_eq!(z, vec![7, 0, 0, 4, 0, 0, 1]);
20//! ```
21//!
22//! # 計算量
23//!
24//! - [`z_algo`][]: $O(n)$
25
26/// 長さ $n$ の列 `s` の Z 配列を計算する。仕様は [crate レベルドキュメント](self) を参照。
27///
28/// # 例
29///
30/// ```
31/// use z_algo::z_algo;
32/// let z = z_algo("aaa".as_bytes());
33/// assert_eq!(z, vec![3, 2, 1]); // z[1] = "aa" と "aaa" の共通接頭辞長
34/// ```
35///
36/// # 計算量
37///
38/// $O(n)$
39pub fn z_algo<T: Eq>(s: &[T]) -> Vec<usize> {
40    if s.is_empty() {
41        return Vec::new();
42    }
43    let n = s.len();
44    let mut z = vec![0; n];
45    z[0] = n;
46    let mut left = 1;
47    let mut right = left;
48    while left < n {
49        while right < n && s[right - left] == s[right] {
50            right += 1;
51        }
52        z[left] = right - left;
53        if left == right {
54            left += 1;
55            right += 1;
56            continue;
57        }
58        let mut next = left + 1;
59        while next < n && next + z[next - left] < right {
60            z[next] = z[next - left];
61            next += 1;
62        }
63        left = next;
64    }
65    z
66}

68#[cfg(test)]
69mod tests {
70    use super::*;
71    use rand::prelude::*;
72    use std::iter;
73    use test_case::test_case;
74
75    fn z_brute<T: Ord>(s: &[T]) -> Vec<usize> {
76        let n = s.len();
77        (0..n)
78            .map(|i| {
79                s.iter()
80                    .zip(s[i..].iter())
81                    .take_while(|&(c, d)| c == d)
82                    .count()
83            })
84            .collect()
85    }

87    #[allow(clippy::unused_unit)]
88    #[test_case("abcabca")]
89    #[test_case("abracadabra")]
90    #[test_case("mississippi")]
91    fn test_hand(s: &str) {
92        let expected = z_brute(s.as_bytes());
93        let result = z_algo(s.as_bytes());
94        assert_eq!(expected, result);
95    }

98    fn test_random() {

102            let s = iter::repeat_with(|| rng.sample(rand::distributions::Alphanumeric))

104                .take(n)
105                .collect::<String>();

109            let result = z_algo(s.as_bytes());
