1//! 素数 $p$ を法とした平方根（modular square root）を、Cipolla のアルゴリズムで求める。
2//!
3//! $a$ が平方非剰余となるような $b$ を探し、2 次拡大体 $\mathbb{F}_p[\sqrt{b^2 - a}]$ を構成する。
4//! この体上で $(b + \sqrt{b^2 - a})^{(p+1)/2}$ を計算すると、$b^2 - a$ が非剰余であることから
5//! フロベニウス写像の性質により虚部が消え、実部として $\sqrt a$ が得られる。
6//! Tonelli–Shanks 法と異なり $p - 1$ の 2-進付値を扱わない分、実装が単純。
7//!
8//! # 仕様
9//!
10//! [`cipolla_sqrt`]`(a, p)` は $p$ を素数として、$x^2 \equiv a \pmod p$ を満たす $x$ を返す。
11//! 平方剰余が存在しなければ `None`。
12//!
13//! # 例
14//!
15//! ```
16//! use cipolla::cipolla_sqrt;
17//! let x = cipolla_sqrt(5u64, 41).unwrap();
18//! assert_eq!(x * x % 41, 5);
19//! assert_eq!(cipolla_sqrt(3u64, 5), None); // 3 は mod 5 の平方非剰余
20//! ```
21//!
22//! # 計算量
23//!
24//! - [`cipolla_sqrt`][]: 期待 $O(\log p)$（非剰余の $b$ を見つける試行回数は期待 $O(1)$ 回）

32/// $x^2 \equiv a \pmod p$ を満たす $x$ を Cipolla のアルゴリズムで求める。
33///
34/// `p` は素数を仮定する。$a$ が平方非剰余の場合は `None`。
35///
36/// # 例
37///
38/// ```
39/// use cipolla::cipolla_sqrt;
40/// let x = cipolla_sqrt(2u64, 7).unwrap();
41/// assert_eq!(x * x % 7, 2);
42/// ```
43///
44/// # 計算量
45///
46/// 期待 $O(\log p)$
47pub fn cipolla_sqrt<T: Unsigned>(a: T, p: T) -> Option<T> {
48    let a = a % p;
49    if p == T::TWO {
50        Some(a)
51    } else if a == T::ZERO {
52        Some(T::ZERO)
53    } else if binary_exponentiation(a, p / T::TWO, T::ONE, |x, y| x * y % p) != T::ONE {
54        None
55    } else {
56        let mut b = T::ZERO;
57        loop {
58            let sqgen = (b * b + p - a) % p;
59            if binary_exponentiation(sqgen, p / T::TWO, T::ONE, |x, y| x * y % p) != T::ONE {
60                return Some(
61                    binary_exponentiation(
62                        [b, T::ONE],
63                        (p + T::ONE) / T::TWO,
64                        [T::ONE, T::ZERO],
65                        |x, y| {
66                            [
67                                (x[0] * y[0] + (x[1] * y[1]) % p * sqgen) % p,
68                                (x[0] * y[1] + x[1] * y[0]) % p,
69                            ]
70                        },
71                    )[0],
72                );
73            }
74            b = b + T::ONE;
75        }
76    }
77}

79// 繰り返し二乗法により、モノイド演算 `f` に関する `a` の `n` 乗を $O(\log n)$ で計算する。
80fn binary_exponentiation<T: Unsigned, U: Copy>(
81    mut a: U,
82    mut n: T,
83    id: U,
84    mut f: impl FnMut(U, U) -> U,
85) -> U {
86    let mut ans = id;
87    while n != T::ZERO {
88        if n % T::TWO == T::ONE {
89            ans = f(ans, a);
90        }
91        a = f(a, a);
92        n = n / T::TWO;
93    }
94    ans
95}

97/// [`cipolla_sqrt`] が要求する、`p` を法とした四則演算に必要な符号なし整数の性質。
98///
99/// # 例
100///
101/// ```
102/// use cipolla::Unsigned;
103/// assert_eq!(u64::ZERO, 0);
104/// assert_eq!(u64::ONE, 1);
105/// assert_eq!(u64::TWO, 2);
106/// ```
107pub trait Unsigned:
108    Sized
109    + Clone
110    + Copy
111    + PartialEq
112    + Add<Output = Self>
113    + Sub<Output = Self>
114    + Mul<Output = Self>
115    + Div<Output = Self>
116    + Rem<Output = Self>
117{
118    /// $0$
119    const ZERO: Self;
120    /// $1$
121    const ONE: Self;
122    /// $2$
123    const TWO: Self;
124}
125
126macro_rules! impl_unsigned {
127    ($($T:ty),+) => {$(
128        impl Unsigned for $T {
129            const ZERO: Self = 0;
130            const ONE: Self = 1;
131            const TWO: Self = 2;
132        }
133    )+}
134}
135impl_unsigned! {u8, u16, u32, u64, u128, usize}

137#[cfg(test)]
138mod tests {
139    use super::cipolla_sqrt;
140
141    #[test]
142    fn test_sqrt_all() {
143        let mut sieve = [false; 100];
144        let sieve_len = sieve.len();
145        for p in 2..sieve_len {
146            if sieve[p] {
147                continue;
148            }
149            // 素数 p を順にテストです。
150            let mut count = 0;
151            for y in 1..p {
152                if let Some(x) = cipolla_sqrt(y, p) {
153                    assert_eq!(x * x % p, y);
154                    count += 1;
155                }
156            }
157            // 平方剰余の個数は ceil(p / 2) 個
158            assert_eq!(count, p / 2);
159            (p * p..)
160                .step_by(p)
161                .take_while(|&q| q < sieve_len)
162                .for_each(|q| sieve[q] = true);
163        }
164    }
165}
