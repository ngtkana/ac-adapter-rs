9pub fn deep_free<O: LazyOps>(root: *mut Node<O>) {
10    if !root.is_null() {
11        unsafe {
12            deep_free((*root).left);
13            deep_free((*root).right);
14            Box::from_raw(root);
15        }
16    }
17}

19/// # Safety notes
20///
21/// `root` は生ポインタで受け渡しします。スプレー操作は祖先へ辿ってから元のノードへ戻って
22/// 書き換えることがあるため、`&mut Node<O>` を引数に取ると（呼び出し期間中ずっと有効な排他参照
23/// という強い保証のもとで）別名経由の書き込みと衝突し、未定義動作になります（Stacked/Tree
24/// Borrows で検証済み）。そのため、この関数とその内部で呼ぶ `splay`/`rotate` は生ポインタのみを
25/// 扱い、フィールドアクセスの瞬間だけ一時的に参照を作ります。
26pub fn access_index<O: LazyOps>(root: *mut Node<O>, mut i: usize) -> *mut Node<O> {
27    let mut root = root;
28    loop {
29        unsafe { (*root).push() };
30        let left = unsafe { (*root).left };
31        if let Some(left) = unsafe { left.as_mut() } {
32            left.push();
33        }
34        let right = unsafe { (*root).right };
35        if let Some(right) = unsafe { right.as_mut() } {
36            right.push();
37        }
38        let lsize = unsafe { left.as_ref() }.map_or(0, |left| left.len);
39        root = match i.cmp(&lsize) {
40            Ordering::Less => left,
41            Ordering::Equal => {
42                splay(root);
43                return root;
44            }
45            Ordering::Greater => {
46                i -= lsize + 1;
47                right
48            }
49        };
50    }
51}

53pub fn merge<O: LazyOps>(left: *mut Node<O>, right: *mut Node<O>) -> *mut Node<O> {
54    if left.is_null() {
55        return right;
56    }
57    if right.is_null() {
58        return left;
59    }
60    let left = access_index(left, unsafe { (*left).len } - 1);
61    unsafe {
62        (*left).push();
63        (*left).right = right;
64        (*right).parent = left;
65        (*left).update();
66    }
67    left
68}

70pub fn split_at<O: LazyOps>(root: *mut Node<O>, at: usize) -> [*mut Node<O>; 2] {
71    if root.is_null() {
72        return [null_mut(), null_mut()];
73    }
74    let len = unsafe { (*root).len };
75    if at == len {
76        [root, null_mut()]
77    } else if at == 0 {
78        [null_mut(), root]
79    } else {
80        let root = access_index(root, at);
81        unsafe { (*root).push() };
82        let left = replace(unsafe { &mut (*root).left }, null_mut());
83        if let Some(left) = unsafe { left.as_mut() } {
84            left.parent = null_mut();
85            unsafe { (*root).update() };
86        }
87        [left, root]
88    }
89}

91/// ノード `x` を根まで splay します。生ポインタで実装している理由は [`access_index`] を参照してください。
92fn splay<O: LazyOps>(x: *mut Node<O>) {
93    loop {
94        let p = unsafe { (*x).parent };
95        if p.is_null() {
96            return;
97        }
98        let g = unsafe { (*p).parent };
99        if !g.is_null() {
100            let x_is_p_left = ptr::eq(x, unsafe { (*p).left });
101            let p_is_g_left = ptr::eq(p, unsafe { (*g).left });
102            if x_is_p_left == p_is_g_left {
103                rotate(p);
104            } else {
105                rotate(x);
106            }
107        }
108        rotate(x);
109    }
110}

112/// `x` をその親 `p` の位置まで回転させます。生ポインタで実装している理由は [`access_index`] を参照してください。
113fn rotate<O: LazyOps>(x: *mut Node<O>) {
114    let p = unsafe { (*x).parent };
115    let g = unsafe { (*p).parent };
116    unsafe { (*x).push() };
117    if ptr::eq(x, unsafe { (*p).left }) {
118        let xr = unsafe { (*x).right };
119        unsafe { (*p).left = xr };
120        if let Some(c) = unsafe { xr.as_mut() } {
121            c.parent = p;
122        }
123        unsafe { (*x).right = p };
124    } else {
125        let xl = unsafe { (*x).left };
126        unsafe { (*p).right = xl };
127        if let Some(c) = unsafe { xl.as_mut() } {
128            c.parent = p;
129        }
130        unsafe { (*x).left = p };
131    }
132    unsafe {
133        (*p).parent = x;
134        (*x).parent = g;
135    }
136    if let Some(g) = unsafe { g.as_mut() } {
137        if ptr::eq(p, g.left) {
138            g.left = x;
139        } else {
140            g.right = x;
141        }
142    }
143    unsafe {
144        (*p).update();
145        (*x).update();
146    }
147}

149pub struct Node<O: LazyOps> {
150    pub left: *mut Self,
151    pub right: *mut Self,
152    pub parent: *mut Self,
153    pub len: usize,
154    pub rev: bool,
155    pub value: O::Value,
156    pub acc: O::Acc,
157    pub lazy: Option<O::Lazy>,
158}
159impl<O: LazyOps> Node<O> {
160    pub fn new(value: O::Value) -> Self {
161        Node {
162            left: null_mut(),
163            right: null_mut(),
164            parent: null_mut(),
165            len: 1,
166            rev: false,
167            acc: O::proj(&value),
168            value,
169            lazy: None,
170        }
171    }
172
173    pub fn dump(&self)
174    where
175        O::Value: Debug,
176        O::Acc: Debug,
177        O::Lazy: Debug,
178    {
179        if let Some(left) = unsafe { self.left.as_ref() } {
180            left.dump();
181        }
182        println!(
183            "{:?}: parent = {:?},  left = {:?}, right = {:?}, len = {}, rev = {}, value = {:?}, \
184             acc = {:?}, lazy = {:?}",
185            std::ptr::from_ref(self),
186            self.parent,
187            self.left,
188            self.right,
189            self.len,
190            self.rev,
191            self.value,
192            self.acc,
193            self.lazy
194        );
195        if let Some(right) = unsafe { self.right.as_ref() } {
196            right.dump();
197        }
198    }
199
200    pub fn update(&mut self) {
201        self.len = 1;
202        self.acc = O::proj(&self.value);
203        if let Some(left) = unsafe { self.left.as_mut() } {
204            left.push();
205            self.len += left.len;
206            self.acc = O::op(&left.acc, &self.acc);
207        }
208        if let Some(right) = unsafe { self.right.as_mut() } {
209            right.push();
210            self.len += right.len;
211            self.acc = O::op(&self.acc, &right.acc);
212        }
213    }
214
215    pub fn push(&mut self) {
216        if let Some(lazy) = self.lazy.take() {
217            O::act_value(&lazy, &mut self.value);
218            O::act_acc(&lazy, &mut self.acc);
219            if let Some(left) = unsafe { self.left.as_mut() } {
220                O::compose_to_option(&lazy, &mut left.lazy);
221            }
222            if let Some(right) = unsafe { self.right.as_mut() } {
223                O::compose_to_option(&lazy, &mut right.lazy);
224            }
225        }
226        if replace(&mut self.rev, false) {
227            std::mem::swap(&mut self.left, &mut self.right);
228            if let Some(left) = unsafe { self.left.as_mut() } {
229                left.rev ^= true;
230            }
231            if let Some(right) = unsafe { self.right.as_mut() } {
232                right.rev ^= true;
233            }
234        }
235    }
236}
