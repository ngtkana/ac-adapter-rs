1//! 文字列の全中心について最長回文の半径を線形時間で求める（Manacher's algorithm）。
2//!
3//! 中心ごとに愚直に回文を伸ばすと $O(n^2)$ かかるが、確定済みの回文の対称性を利用すると
4//! 多くの中心で伸長処理を省略できる。中心 $i$ が既知の回文の内側にあれば、対称な中心の
5//! 結果をそのまま折り返して使え、外側にはみ出す部分だけ愚直に文字比較すればよい。
6//! この比較回数が amortized で定数回に抑えられるため、全体で $O(n)$ になる。
7//!
8//! # 仕様
9//!
10//! [`manacher`] が唯一の公開関数。文字列 $s$（長さ $n$）を受け取り、長さ $2n + 1$ の
11//! 配列 $A$ を返す。
12//!
13//! $$
14//! A_i = \max \left\{ r - l \;\middle|\; 0 \le l \le r \le n,\ l + r = i,\ s[l..r] \text{ は回文} \right\}
15//! $$
16//!
17//! $i$ が偶数のとき中心は文字 $s_{i / 2}$ 上、奇数のとき文字と文字の間にある。
18//!
19//! # 例
20//!
21//! ```
22//! use manacher::manacher;
23//! let s = "mississippi";
24//! let a = manacher(s.as_bytes());
25//! assert_eq!(a, vec![
26//!     0, 1, 0, 1, 0, 1, 4, 1, 0, 7, 0, 1, 4, 1, 0, 1, 0, 1, 4, 1, 0, 1, 0
27//! ]);
28//! ```
29//!
30//! # 計算量
31//!
32//! - [`manacher`][]: $O(n)$
33
34/// 文字列 $s$ の全中心について、最長回文の半径を格納した配列を返す。
35///
36/// 中心ごとに愚直に伸ばすと $O(n^2)$ かかるが、確定済みの回文の対称性を利用して
37/// 多くの中心の計算を省略し、$O(n)$ で求める。
38///
39/// # 仕様
40///
41/// $n = $ `s.len()` として、長さ $2n + 1$ の配列 $A$ を返す。
42///
43/// $$
44/// A_i = \max \left\{ r - l \;\middle|\; 0 \le l \le r \le n,\ l + r = i,\ s[l..r] \text{ は回文} \right\}
45/// $$
46///
47/// $i$ が偶数のとき中心は文字 $s_{i / 2}$ 上、奇数のとき文字と文字の間にある。
48///
49/// # 例
50///
51/// $s = $ "mississippi" のとき、中心 $i$ ごとの最長回文の半径 $A_i$ は次の通り
52/// （`_` は文字と文字の間の中心）。
53///
54/// | $i$ | 中心 | $A_i$ | 対応する回文 |
55/// | --- | --- | --- | --- |
56/// | 0 | (前端) | 0 | |
57/// | 1 | m | 1 | m |
58/// | 2 | _ | 0 | |
59/// | 3 | i | 1 | i |
60/// | 4 | _ | 0 | |
61/// | 5 | s | 1 | s |
62/// | 6 | _ | 4 | issi |
63/// | 7 | s | 1 | s |
64/// | 8 | _ | 0 | |
65/// | 9 | i | 7 | ississi |
66/// | 10 | _ | 0 | |
67///
68/// ```
69/// use manacher::manacher;
70/// let s = "mississippi";
71/// let a = manacher(s.as_bytes());
72/// assert_eq!(a, vec![
73///     0, 1, 0, 1, 0, 1, 4, 1, 0, 7, 0, 1, 4, 1, 0, 1, 0, 1, 4, 1, 0, 1, 0
74/// ]);
75/// ```
76///
77/// # 計算量
78///
79/// $O(n)$
80pub fn manacher<T: Eq>(s: &[T]) -> Vec<usize> {
81    let n = s.len();
82    let mut a = vec![0; 2 * n + 1];
83    let mut i = 1;
84    let mut j = 1;
85    while i <= 2 * n {
86        while j < i && i + j < 2 * n && s[(i - j) / 2 - 1] == s[usize::midpoint(i, j)] {
87            j += 2;
88        }
89        a[i] = j;
90        if j == 0 {
91            i += 1;
92            j = 1;
93            continue;
94        }
95        let mut k = 1;
96        while k <= i && k + a[i - k] < j {
97            a[i + k] = a[i - k];
98            k += 1;
99        }
100        i += k;
101        j -= k;
102    }
103    a
104}

106#[cfg(test)]
107mod tests {
108    use super::*;
109    use rand::rngs::StdRng;
110    use rand::Rng;
111    use rand::SeedableRng;
112
113    #[test]
114    fn test_manacher() {
115        let mut rng = StdRng::seed_from_u64(42);
116        for _ in 0..200 {
117            let n = rng.gen_range(0..10);
118            let s = (0..n).map(|_| rng.gen_range(0..4)).collect::<Vec<_>>();
119            let result = manacher(&s);
120            for (i, &result) in result.iter().enumerate() {
121                let mut l = i / 2;
122                let mut r = i - l;
123                while 0 < l && r < n && s[l - 1] == s[r] {
124                    l -= 1;
125                    r += 1;
126                }
127                assert_eq!(result, r - l);
128            }
129        }
130    }
