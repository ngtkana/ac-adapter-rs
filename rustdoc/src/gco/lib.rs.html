1//! 次数 $2$ 以下の劣モジュラなグラフカット最適化問題を解く。
2//!
3//! 任意の擬似ブール関数 $f: \{0, 1\}^n \to \mathbb{R}$ は、多重線形多項式
4//! $f(\boldsymbol{x}) = a + \sum_i a_i x_i + \sum_{i,j} a_{i,j} x_i x_j + \dots$
5//! として一意に表せる。次数が $2$ 以下で、各二次項が劣モジュラ条件
6//! $a_{i,i} + a_{j,j} \le a_{i,j} + a_{j,i}$ を満たすとき、$f$ の最小化は $s$-$t$ 最小カット問題に
7//! 帰着できる。変数ごとに頂点を用意し、一次項をソース・シンクへの辺重みに、二次項を変数間の辺重みに
8//! 変換することでこれを実現し、最小カットの計算には [`dinic`] を用いる。
9//!
10//! # 仕様
11//!
12//! - [`Gco::unary`] で一次項 $c_0 (1 - x_i) + c_1 x_i$ を、[`Gco::binary`] で二次項
13//!   $c_{0,0}(1-x_i)(1-x_j) + c_{0,1}(1-x_i)x_j + c_{1,0}x_i(1-x_j) + c_{1,1}x_ix_j$ を追加する
14//! - 二次項は劣モジュラ条件 $c_{0,0} + c_{1,1} \le c_{0,1} + c_{1,0}$ を満たす必要がある（違反時は panic）
15//! - コストの型は [`i64`]
16//! - [`Gco::solve`] で最小値と、それを達成する $x$（`false` は $0$、`true` は $1$）を得る
17//! - 変数の反転（$x_i \mapsto 1 - x_i$）は自動で行われない
18//!
19//! # 例
20//!
21//! $f(x, y) = (10 + 10x) + (40 - 30y) + 99(x + y - 2xy)$ は $(x, y) = (1, 1)$ で最小値 $30$ を取る。
22//!
23//! ```
24//! use gco::Gco;
25//!
26//! let mut gco = gco::Gco::new(2);
27//! gco.unary(0, [10, 20]);
28//! gco.unary(1, [40, 10]);
29//! gco.binary([0, 1], [[0, 99], [99, 0]]);
30//!
31//! let result = gco.solve();
32//! assert_eq!(result.value, 30);
33//! assert_eq!(&result.args, &[true, true]);
34//! ```
35//!
36//! # 計算量
37//!
38//! - [`Gco::solve`][]: $O(n^2 m)$（$n$ は頂点数、$m$ は辺数。[`dinic`] の最大流計算量に従う）
39
40use dinic::Dinic;
41use std::cmp::Ordering;
42
43/// グラフカット最適化問題のソルバー。
44///
45/// [`Gco::new`] で変数の個数を指定して初期化し、[`Gco::unary`], [`Gco::binary`] で
46/// $f$ に項を追加してから [`Gco::solve`] で最小値と最小点を求める。
47#[derive(Clone, Debug, Default, Hash, PartialEq)]
48pub struct Gco {
49    vars: usize,
50    unary: Vec<Unary>,
51    binary: Vec<Binary>,
52}
53impl Gco {
54    /// $n$ 個の変数を持つソルバーを初期化する。項は何も追加されていない状態（$f \equiv 0$）から始まる。
55    pub fn new(n: usize) -> Self {
56        Self {
57            vars: n,
58            ..Self::default()
59        }
60    }
61
62    /// 一次項 $c_0 (1 - x_i) + c_1 x_i$ を $f$ に加える。
63    ///
64    /// # 例
65    ///
66    /// ```
67    /// # use gco::Gco;
68    /// # let mut gco = gco::Gco::new(2);
69    /// gco.unary(0, [0, 10]); // 10 x_0
70    /// gco.unary(1, [-40, 0]); // -40 (1 - x_1)
71    /// ```
72    pub fn unary(&mut self, i: usize, cost: [i64; 2]) {
73        self.unary.push(Unary { i, cost });
74    }
75
76    /// 二次項
77    /// $$
78    /// c_{0,0}(1-x_i)(1-x_j) + c_{0,1}(1-x_i)x_j + c_{1,0}x_i(1-x_j) + c_{1,1}x_ix_j
79    /// $$
80    /// を $f$ に加える。
81    ///
82    /// # panics
83    ///
84    /// 劣モジュラ条件 $c_{0,0} + c_{1,1} \le c_{0,1} + c_{1,0}$ を満たさないとき。
85    ///
86    /// # 例
87    ///
88    /// ```
89    /// # use gco::Gco;
90    /// # let mut gco = gco::Gco::new(2);
91    /// gco.binary([0, 1], [[0, 10], [0, 0]]); // x_0 = 0, x_1 = 1 のとき 10
92    /// ```
93    pub fn binary(&mut self, ij: [usize; 2], cost: [[i64; 2]; 2]) {
94        assert!(
95            is_submodular(cost),
96            "The cost should be submodular. cost = {cost:?}"
97        );
98        self.binary.push(Binary { ij, cost });
99    }
100
101    /// $f$ の最小値とそれを達成する $x$ を返す。
102    ///
103    /// # 計算量
104    ///
105    /// $O(n^2 m)$（$n$ は頂点数、$m$ は辺数）
106    pub fn solve(&self) -> GcoResult {
107        solve(self)
108    }
109}
110
111/// [`Gco::solve`] の結果。$f$ の最小値と、それを達成する $x$ を保持する。
112///
113/// - $x_i = 0 \Leftrightarrow$ `args[i] == false`
114/// - $x_i = 1 \Leftrightarrow$ `args[i] == true`
115#[derive(Clone, Debug, Default, Hash, PartialEq, Eq)]
116pub struct GcoResult {
117    /// $f$ の最小値。
118    pub value: i64,
119    /// 最小値を達成する $x$。
120    pub args: Vec<bool>,
121}
122
123fn solve(gco: &Gco) -> GcoResult {
124    let Gco {
125        vars,
126        unary,
127        binary,
128    } = gco;
129    let s = *vars;
130    let t = s + 1;
131    let vertex_count = t + 1;
132    let mut constant = 0;
133    let mut dinic = Dinic::<i64>::new(vertex_count);
134    for &binary in binary {
135        let Binary { ij, mut cost } = binary;
136        let d = cost[0][1] + cost[1][0] - cost[0][0] - cost[1][1];
137        match d.cmp(&0) {
138            Ordering::Less => unreachable!(),
139            Ordering::Greater => {
140                dinic.add_edge(ij[0], ij[1], d);
141                cost[0][1] -= d;
142            }
143            Ordering::Equal => (),
144        }
145        for (p, &ij) in ij.iter().enumerate() {
146            let d = diff(cost, p);
147            match d.cmp(&0) {
148                Ordering::Less => {
149                    constant += d;
150                    dinic.add_edge(ij, t, -d);
151                }
152                Ordering::Greater => {
153                    dinic.add_edge(s, ij, d);
154                }
155                Ordering::Equal => (),
156            }
157            for (i, j) in (0..2)
158                .flat_map(|i| (0..2).map(move |j| (i, j)))
159                .filter(|&(i, j)| [i, j][p] == 1)
160            {
161                cost[i][j] -= d;
162            }
163        }
164        let d = cost[0][0];
165        cost.iter_mut().flatten().for_each(|x| *x -= d);
166        constant += d;
167        assert_eq!(cost, [[0; 2]; 2]);
168    }
169    for &unary in unary {
170        let cost = unary.cost;
171        match cost[0].cmp(&cost[1]) {
172            Ordering::Less => {
173                dinic.add_edge(s, unary.i, cost[1] - cost[0]);
174                constant += cost[0];
175            }
176            Ordering::Greater => {
177                dinic.add_edge(unary.i, t, cost[0] - cost[1]);
178                constant += cost[1];
179            }
180            Ordering::Equal => constant += cost[0],
181        }
182    }
183    let value = constant + dinic.flow(s, t);
184    let args = dinic.min_cut(s)[..*vars]
185        .iter()
186        .map(|&b| !b)
187        .collect::<Vec<_>>();
188    GcoResult { value, args }
189}

191#[derive(Clone, Debug, Default, Hash, PartialEq, Copy)]
192struct Unary {
193    i: usize,
194    cost: [i64; 2],
195}
196#[derive(Clone, Debug, Default, Hash, PartialEq, Copy)]
197struct Binary {
198    ij: [usize; 2],
199    cost: [[i64; 2]; 2],
200}
201
202fn index_by_array2(cost: [[i64; 2]; 2], index: [usize; 2]) -> i64 {
203    cost[index[0]][index[1]]
204}
205fn diff(cost: [[i64; 2]; 2], p: usize) -> i64 {
206    assert!(is_submodular(cost));
207    let mut index = [0, 0];
208    index[p] = 1;
209    index_by_array2(cost, index) - cost[0][0]
210}
211fn is_submodular(cost: [[i64; 2]; 2]) -> bool {
212    cost[0][0] + cost[1][1] <= cost[0][1] + cost[1][0]
213}

227    fn brute(gco: &Gco) -> i64 {
228        (0..1 << gco.vars)
229            .map(|bs| {
230                gco.unary
231                    .iter()
232                    .map(|&Unary { i, cost }| cost[bs >> i & 1])
233                    .chain(
234                        gco.binary
235                            .iter()
236                            .map(|&Binary { ij: [i, j], cost }| cost[bs >> i & 1][bs >> j & 1]),
237                    )
238                    .sum::<i64>()
239            })
240            .min()
241            .unwrap()
242    }

244    #[test]
245    fn test_random() {
246        let mut rng = StdRng::seed_from_u64(42);
247        for _ in 0..200 {
248            let n = 3;
249            let mut gco = Gco::new(n);
250            let unary_count = 0;
251            let binary_count = 1;
252            for _ in 0..unary_count {
253                let i = rng.gen_range(0..n);
254                let x = rng.gen_range(-9..10);
255                let y = rng.gen_range(-9..10);
256                gco.unary(i, [x, y]);
257            }
258            for _ in 0..binary_count {
259                let (i, j) = rng.sample(DistinctTwo(0..n));
260                let ij = [i, j];
261                let cost = loop {
262                    let cost = [[rng.gen_range(-9..10), rng.gen_range(-9..10)], [
263                        rng.gen_range(-9..10),
264                        rng.gen_range(-9..10),
265                    ]];
266                    if is_submodular(cost) {
267                        break cost;
268                    }
269                };
270                gco.binary(ij, cost);
271            }
272
273            let expected = brute(&gco);
274            let GcoResult { value, args } = gco.clone().solve();
275            assert_eq!(expected, value);
276            assert_eq!(
277                value,
278                gco.unary
279                    .iter()
280                    .map(|&Unary { i, cost }| cost[usize::from(args[i])])
281                    .chain(gco.binary.iter().map(|&Binary { ij: [i, j], cost }| cost
282                        [usize::from(args[i])][usize::from(args[j])]),)
283                    .sum::<i64>()
284            );
285        }
286    }
