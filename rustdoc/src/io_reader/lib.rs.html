1//! 空白区切りの標準入力・文字列を型安全にパースする。
2//!
3//! `input!` マクロの右辺には型ではなく [`Parser`] trait を実装した**値**を書き、それらを組み合わせて
4//! 複合的なパーサーを作る。入力元は [`BufRead`] を実装した型を [`Source`] でラップして与え、
5//! 省略時は標準入力（[`stdin_source`]）、テストでは文字列（[`Source::from`]）を使うのが典型的。
6//!
7//! # 仕様
8//!
9//! - `input! { pat = parser, ... }`: 標準入力から読み取り `pat` に束縛
10//! - `input! { from expr, pat = parser, ... }`: `expr`（[`Source`] に変換可能な値）から読み取り
11//! - パーサーの種類:
12//!   - 基本型: [`Char`], [`Str`], [`I32`], [`U64`] など（`Canonical<P>` により [`FromStr`] へ委譲）
13//!   - 列: [`Vector<P>`] で長さ $n$ の `Vec`
14//!   - 配列: [`Array<N, P>`] で長さ $N$ の配列
15//!   - タプル: `(P0, P1, ...)` で複数要素をまとめて読み取り
16//!   - 特殊: [`Usize1`]（1-indexed → 0-indexed）、[`Bools<Z, O>`]（0/1 文字列）、[`Bytes`]（バイト列）、[`Expect`]（固定トークンの検証）
17//!
18//! # 例
19//!
20//! ```
21//! use io_reader::{input, Usize, I32, Vector};
22//!
23//! input! {
24//! #   from "2 3 10 20 30",
25//!     n = Usize,
26//!     m = Usize,
27//!     a = Vector(I32, 3),
28//! }
29//! # assert_eq!(n, 2);
30//! # assert_eq!(m, 3);
31//! # assert_eq!(a, [10, 20, 30]);
32//! ```
33
34use std::{
35    fmt::Debug,
36    io::{BufRead, BufReader, Stdin, stdin},
37    marker::PhantomData,
38    str::{FromStr, SplitWhitespace},
39    sync::{Mutex, MutexGuard, OnceLock},
40};
41
42static STDIN_SOURCE: OnceLock<Mutex<Source<BufReader<Stdin>>>> = OnceLock::new();
43
44/// プロセス全体で共有する標準入力の [`Source`] を取得する。
45///
46/// 実体は [`OnceLock<Mutex<_>>`] に包まれた `static` で、初回呼び出し時に遅延初期化する。
47/// 返り値は [`MutexGuard`] なので、drop されるまでロックを保持する。
48pub fn stdin_source() -> MutexGuard<'static, Source<BufReader<Stdin>>> {
49    STDIN_SOURCE
50        .get_or_init(|| Mutex::new(Source::new(BufReader::new(stdin()))))
51        .lock()
52        .unwrap()
53}

55/// 入力をパースして変数に束縛する。
56///
57/// 右辺には型ではなく [`Parser`] の値を書く。`pat = parser` をコンマ区切りで並べると、書いた順に
58/// 読み取って束縛する。`from` 節を省略すると標準入力（[`stdin_source`]）から読み取る。
59///
60/// # 仕様
61///
62/// - `input! { pat = parser, ... }`: 標準入力から読み取り
63/// - `input! { from expr, pat = parser, ... }`: `expr`（[`Source`] に変換可能な値）から読み取り
64///
65/// # 例
66///
67/// ```
68/// use io_reader::{input, U32, Str};
69///
70/// input! {
71///     from "42 abc",
72///     x = U32,
73///     s = Str,
74/// }
75/// assert_eq!(x, 42);
76/// assert_eq!(s, "abc");
77/// ```
78#[macro_export]
79macro_rules! input {
80    (@from [$source:expr] @rest $($pat:pat = $parser:expr),* $(,)?) => {
81        $(
82            let $pat = $crate::read_value!(@from [$source] @rest $parser);
83        )*
84    };
85
86    (from $str:expr, $($rest:tt)*) => {
87        let mut source = $crate::Source::from($str);
88        $crate::input! {
89            @from [&mut source]
90            @rest $($rest)*
91        }
92    };
93    ($($rest:tt)*) => {
94        let mut source = $crate::stdin_source();
95        $crate::input! {
96            @from [&mut *source]
97            @rest $($rest)*
98        }
99        drop(source)
100    };
101}

103/// 単一の値を 1 回だけパースする。
104///
105/// `input!` マクロが内部で使う 1 回分の読み取り処理だが、単発の読み取りにも直接使える。
106///
107/// # 仕様
108///
109/// - `read_value! { @from [source_expr] parser }`: 指定した [`Source`] から読み取り
110/// - `read_value! { parser }`: 標準入力（[`stdin_source`]）から読み取り
111///
112/// # 例
113///
114/// ```
115/// use io_reader::{read_value, Source, U32};
116///
117/// let mut source = Source::from("12");
118/// let x = read_value! { @from [&mut source] U32 };
119/// assert_eq!(x, 12);
120/// ```
121#[macro_export]
122macro_rules! read_value {
123    (@from [$source:expr] $(,)? @rest $parser:expr) => {
124        $crate::Parser::read(&$parser, $source)
125    };
126
127    // interface
128    {
129        @from [$source:expr] $(,)?
130        $parser:expr
131    } => {
132        read_value!(@from [$source] @rest $parser)
133    };
134    {
135        $parser:expr
136    } => {
137        {
138            let mut source = $crate::stdin_source();
139            read_value!(@from [&mut source] @rest $parser)
140        }
141    };
142}

144/// [`BufRead`] をラップし、空白区切りの token を順に取り出す。
145///
146/// 内部で行を読み込みながら `split_whitespace` するため、改行をまたいでも連続した token 列として扱う。
147///
148/// # 例
149///
150/// ```
151/// use io_reader::Source;
152///
153/// let mut source = Source::from("0120 444 444");
154///
155/// assert_eq!(source.next_token_unwrap(), "0120");
156/// assert_eq!(source.next_token_unwrap(), "444");
157/// assert_eq!(source.next_token_unwrap(), "444");
158/// assert_eq!(source.next_token(), None);
159/// ```
160#[derive(Debug)]
161pub struct Source<R: BufRead> {
162    reader: R,
163    line: String,
164    tokens: SplitWhitespace<'static>,
165}
166
167impl<R: BufRead> Source<R> {
168    /// [`BufRead`] から [`Source`] を構築する。
169    ///
170    /// # 例
171    ///
172    /// ```
173    /// use io_reader::Source;
174    /// let mut source = Source::from("hello world");
175    /// ```
176    pub fn new(reader: R) -> Self {
177        let line = String::new();
178        let tokens = unsafe {
179            std::mem::transmute::<std::str::SplitWhitespace<'_>, std::str::SplitWhitespace<'_>>(
180                line.split_whitespace(),
181            )
182        };
183        Self {
184            reader,
185            line,
186            tokens,
187        }
188    }
189
190    /// 空白区切りで次の token を取得する。入力の終端では `None` を返す。
191    pub fn next_token(&mut self) -> Option<&str> {
192        loop {
193            if let Some(result) = self.tokens.next() {
194                return Some(result);
195            }
196            let mut line = String::new();
197            if self.reader.read_line(&mut line).unwrap() == 0 {
198                return None;
199            }
200            self.line = line;
201            self.tokens = unsafe {
202                std::mem::transmute::<std::str::SplitWhitespace<'_>, std::str::SplitWhitespace<'_>>(
203                    self.line.split_whitespace(),
204                )
205            };
206        }
207    }
208
209    /// [`next_token`](Self::next_token) の panic 版。終端に達すると panic する。
210    pub fn next_token_unwrap(&mut self) -> &str {
211        self.next_token()
212            .unwrap_or_else(|| panic!("unexpected end of input."))
213    }
214}

216impl<'a> From<&'a str> for Source<BufReader<&'a [u8]>> {
217    /// 文字列から [`Source`] を構築する。テストでの入力構築に使う。
218    fn from(value: &'a str) -> Self {
219        Self::new(BufReader::new(value.as_bytes()))
220    }
221}

223/// [`Source`] から 0 個以上の token を読み取り、値へ変換するアルゴリズムを表す。
224///
225/// `input!` マクロの右辺に書く値の型は、すべてこの trait を実装する。
226pub trait Parser: Copy {
227    /// 変換後の値の型。
228    type Output;
229
230    /// [`Source`] から token を読み取り、[`Output`](Self::Output) 型の値に変換する。
231    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output;
232}

234/// [`FromStr`] へ委譲して 1 token を parse する [`Parser`]。
235///
236/// [`Char`], [`Str`], [`I32`] などの基本型パーサーは、すべてこの型のインスタンスとして定義される。
237pub struct Canonical<P>(PhantomData<fn(P)>);
238impl<P> Clone for Canonical<P> {
239    fn clone(&self) -> Self {
240        *self
241    }
242}
243impl<P> Copy for Canonical<P> {}
244
245macro_rules! define_cannonical_parser {
246    ($($prim:ty => $cann:ident),+$(,)?) => {$(
247        /// [`
248        #[doc = stringify!($prim)]
249        /// `] を [`FromStr`] 経由で parse する [`Parser`]。
250        #[allow(non_upper_case_globals)]
251        pub const $cann: Canonical<$prim> = Canonical::<$prim>(PhantomData);
252    )+}
253}
254
255define_cannonical_parser! {
256    char => Char,
257    String => Str,
258    u8 => U8,
259    u16 => U16,
260    u32 => U32,
261    u64 => U64,
262    u128 => U128,
263    usize => Usize,
264    i8 => I8,
265    i16 => I16,
266    i32 => I32,
267    i64 => I64,
268    i128 => I128,
269    isize => Isize,
270}
271
272impl<P: FromStr> Parser for Canonical<P>
273where
274    P::Err: Debug,
275{
276    type Output = P;
277
278    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
279        let token = source.next_token_unwrap();
280        token.parse().unwrap_or_else(|e| {
281            panic!(
282                "failed to parse the token `{token}` to the value of type `{ty}`: {e:?}.",
283                ty = std::any::type_name::<P>()
284            )
285        })
286    }
287}

289/// 固定の token を期待して読み取る [`Parser`]。
290///
291/// [`Source`] から token をちょうど 1 個読み、指定した文字列と一致するかを検証する。
292/// 一致しない場合は panic する。区切り文字や固定フォーマットの検証に使う。
293///
294/// # 例
295///
296/// ```
297/// use io_reader::{input, Expect, U32, Source};
298///
299/// let mut source = Source::from("0 1 -1 2");
300///
301/// input! {
302///     @from [&mut source] @rest
303///     a = U32,
304///     b = U32,
305///     _ = Expect("-1"),
306///     c = U32,
307/// }
308/// ```
309#[derive(Clone, Copy)]
310pub struct Expect(pub &'static str);
311impl Parser for Expect {
312    type Output = ();
313
314    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
315        let token = source.next_token_unwrap();
316        assert_eq!(
317            self.0,
318            token,
319            "Expected a fixed token `{expected}`, but got `{token}`",
320            expected = self.0
321        );
322    }
323}

325/// 1-indexed の `usize` を読み取り、0-indexed に変換する [`Parser`]。
326///
327/// 読み取った値を $x$ とすると $x - 1$ を返す。$x = 0$ の場合は panic する。
328///
329/// # 例
330///
331/// ```
332/// use io_reader::{Usize1, Source, Parser};
333/// # let mut source = Source::from("5");
334/// let result = Usize1.read(&mut source);
335/// assert_eq!(result, 4);  // 5 - 1 = 4
336/// ```
337#[derive(Clone, Copy)]
338pub struct Usize1;
339impl Parser for Usize1 {
340    type Output = usize;
341
342    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
343        Canonical::<usize>(PhantomData)
344            .read(source)
345            .checked_sub(1)
346            .unwrap_or_else(|| panic!("expected `Usize1` but got 0"))
347    }
348}

350/// 長さ $n$ を指定して `Vec<P::Output>` を parse する [`Parser`]。
351///
352/// `P` を $n$ 回連続で適用して集める。
353///
354/// # 例
355///
356/// ```
357/// use io_reader::{input, Vector, I32, Source, Parser};
358///
359/// let mut source = Source::from("1 2 3");
360///
361/// input! {
362///    @from [&mut source] @rest
363///     v = Vector(I32, 3),
364/// }
365///
366/// assert_eq!(v, vec![1, 2, 3]);
367/// ```
368#[derive(Clone, Copy)]
369pub struct Vector<P>(pub P, pub usize);
370impl<P: Parser> Parser for Vector<P> {
371    type Output = std::vec::Vec<P::Output>;
372
373    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
374        std::iter::repeat_with(|| self.0.read(source))
375            .take(self.1)
376            .collect()
377    }
378}

380/// 長さ $N$ の `[P::Output; N]` を parse する [`Parser`]。
381///
382/// `N` はコンパイル時に固定する定数で、`P` を $N$ 回連続で適用する。
383///
384/// # 例
385///
386/// ```
387/// use io_reader::{input, Array, I32, Source, Parser};
388///
389/// let mut source = Source::from("10 20 30");
390///
391/// input! {
392///     @from [&mut source] @rest
393///     arr = Array::<3, _>(I32),
394/// }
395///
396/// assert_eq!(arr, [10, 20, 30]);
397/// ```
398#[derive(Clone, Copy)]
399pub struct Array<const N: usize, P>(pub P);
400impl<const N: usize, P: Parser> Parser for Array<N, P> {
401    type Output = [P::Output; N];
402
403    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
404        std::array::from_fn(|_| self.0.read(source))
405    }
406}

408macro_rules! impl_parser_for_tuples {
409    ($head:ident, $($tail:ident,)*) => {
410        impl<$head: Parser, $($tail: Parser,)*> Parser for ($head, $($tail,)*) {
411            type Output = ($head::Output, $($tail::Output,)*);
412
413            fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
414                #[allow(non_snake_case)]
415                let ($head, $($tail,)*) = self;
416                ($head.read(source), $($tail.read(source),)*)
417            }
418        }
419        impl_parser_for_tuples!{$($tail,)*}
420    };
421    {} => {};
422}
423
424impl_parser_for_tuples! {
425    P0, P1, P2, P3, P4, P5,
426    P6, P7, P8, P9, P10, P11,
427    P12, P13, P14, P15, P16,
428}

430/// 1 token を文字ごとに解釈し `Vec<bool>` として parse する [`Parser`]。
431///
432/// 文字 $Z$ を `false`、文字 $O$ を `true` に対応させる。それ以外の文字が現れると panic する。
433///
434/// # 例
435///
436/// ```
437/// use io_reader::{input, Bools, Source, Parser};
438///
439/// let mut source = Source::from("001101");
440///
441/// input! {
442///     @from [&mut source] @rest
443///     bits = Bools::<'0', '1'>,
444/// }
445///
446/// assert_eq!(bits, vec![false, false, true, true, false, true]);
447/// ```
448#[derive(Clone, Copy)]
449pub struct Bools<const Z: char, const O: char>;
450impl<const Z: char, const O: char> Parser for Bools<Z, O> {
451    type Output = Vec<bool>;
452
453    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
454        source
455            .next_token_unwrap()
456            .chars()
457            .map(|c| {
458                if c == Z {
459                    false
460                } else if c == O {
461                    true
462                } else {
463                    panic!("unexpected charactor {c} in parsing `Bools<{Z}, {O}>`");
464                }
465            })
466            .collect()
467    }
468}

470/// 1 token をそのまま `Vec<u8>` として parse する [`Parser`]。
471///
472/// # 例
473///
474/// ```
475/// use io_reader::{input, Bytes, Source, Parser};
476///
477/// let mut source = Source::from("hello");
478///
479/// input! {
480///     @from [&mut source] @rest
481///     bytes = Bytes,
482/// }
483///
484/// assert_eq!(bytes, [104, 101, 108, 108, 111]);
485/// ```
486#[derive(Clone, Copy)]
487pub struct Bytes;
488impl Parser for Bytes {
489    type Output = Vec<u8>;
490
491    fn read<R: BufRead>(&self, source: &mut Source<R>) -> Self::Output {
492        source.next_token_unwrap().as_bytes().to_vec()
493    }
