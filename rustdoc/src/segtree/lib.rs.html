1//! モノイドを載せた完全二分木（セグメント木）
2//!
3//! 葉に要素を並べ、各内部ノードに左右の子の積を持たせた完全二分木として実装する。
4//! 結合律さえ満たせば区間 $[l, r)$ の総積は $O(\log n)$ 個のノード値の積に分解でき、
5//! 1 点更新も葉から根への経路上 $O(\log n)$ 個のノードを更新するだけで済む。
6//!
7//! # 仕様
8//!
9//! [`Op`] トレイトでモノイド $(S, \cdot, e)$ を定義する。
10//!
11//! * [`Op::identity`]: 単位元 $e$
12//! * [`Op::op`]: 積 $x \cdot y$（結合律を満たすこと）
13//!
14//! [`Sparse2dSegtree`], [`Dense2dSegtree`] で使う場合は、さらに可換律も要求する。
15//!
16//! 構築には 2 種類の API がある。
17//!
18//! * `new`: 長さのみを指定し、単位元で初期化する
19//! * `from_slice`: 初期値のスライスから構築する
20//!
21//! 更新には 2 種類の API がある。
22//!
23//! * `entry` API（[`Segtree`], [`SegtreeWithCompression`], [`Dense2dSegtree`]）: 要素を直接書き換える
24//! * `apply` API（[`Sparse2dSegtree`]）: $f(x \cdot y) = x \cdot f(y)$ を満たす関数 $f$ を 1 要素に適用する
25//!
26//! # 例
27//!
28//! ```
29//! use segtree::Op;
30//! use segtree::Segtree;
31//!
32//! enum Add {}
33//! impl Op for Add {
34//!     type Value = i64;
35//!     fn identity() -> i64 {
36//!         0
37//!     }
38//!     fn op(lhs: &i64, rhs: &i64) -> i64 {
39//!         lhs + rhs
40//!     }
41//! }
42//!
43//! let mut segtree = Segtree::<Add>::from_slice(&[1, 2, 3, 4, 5]);
44//! assert_eq!(segtree.fold(1..4), 2 + 3 + 4);
45//!
46//! *segtree.entry(0) = 10;
47//! assert_eq!(segtree.fold(..), 10 + 2 + 3 + 4 + 5);
48//! ```
49//!
50//! # 計算量
51//!
52//! - 構築（[`Segtree::new`], [`Segtree::from_slice`]）: $O(n)$
53//! - 畳み込み（[`Segtree::fold`]）: $O(\log n)$
54//! - 1 点更新（[`Segtree::entry`]）: $O(\log n)$
55//! - 二分探索（[`Segtree::max_right`], [`Segtree::min_left`]）: $O(\log n)$
56

65/// A trait for segment tree operations.
66pub trait Op {
67    /// The value type.
68    type Value;
69
70    /// Returns the identity value $e$.
71    fn identity() -> Self::Value;
72    /// Multiplies two values: $x \cdot y$.
73    fn op(lhs: &Self::Value, rhs: &Self::Value) -> Self::Value;
74}

76/// A segment tree.
77/// Use [`as_slice()`](Self::as_slice()) to debug.
78pub struct Segtree<O: Op> {
79    len: usize,
80    offset: usize,
81    values: Vec<O::Value>,
82}
83impl<O: Op> Segtree<O> {
84    /// Constructs a new segment tree with the specified length, filled with the identity value.
85    pub fn new(len: usize) -> Self
86    where
87        O::Value: Clone,
88    {
89        let offset = len.next_power_of_two();
90        Self {
91            len,
92            offset,
93            values: vec![O::identity(); 2 * offset],
94        }
95    }
96
97    /// Constructs from the specified values.
98    pub fn from_slice(elms: &[O::Value]) -> Self
99    where
100        O::Value: Clone,
101    {
102        let len = elms.len();
103        let offset = len.next_power_of_two();
104        let mut values = vec![O::identity(); 2 * offset];
105        values[offset..offset + len].clone_from_slice(elms);
106        for i in (1..offset).rev() {
107            values[i] = O::op(&values[i * 2], &values[i * 2 + 1]);
108        }
109        Self {
110            len,
111            offset,
112            values,
113        }
114    }
115
116    /// Returns $x_l \cdot x_{l+1} \cdot \ldots \cdot x_{r-1}$.
117    pub fn fold<R: RangeBounds<usize>>(&self, range: R) -> O::Value {
118        let Self {
119            len,
120            offset,
121            ref values,
122        } = *self;
123        let (mut start, mut end) = open(range, len);
124        assert!(start <= end && end <= len);
125        start += offset;
126        end += offset;
127        let mut left = O::identity();
128        let mut right = O::identity();
129        while start < end {
130            if start % 2 == 1 {
131                left = O::op(&left, &values[start]);
132                start += 1;
133            }
134            if end % 2 == 1 {
135                end -= 1;
136                right = O::op(&values[end], &right);
137            }
138            start /= 2;
139            end /= 2;
140        }
141        O::op(&left, &right)
142    }
143
144    /// Returns the maximum $r$ s.t. $f(x _ l \cdot \dots \cdot x _ { r - 1 })$ is `true`, if $f$ is monotone.
145    ///
146    /// More precisely, $r$ satisfies:
147    ///
148    /// 1. $f(x _ l \cdot \dots \cdot x _ { r - 1 })$ is `true`, or $l = r$
149    /// 2. $f(x _ l \cdot \dots \cdot x _ { r } )$ is `false`, or $r = n$
150    pub fn max_right(&self, start: usize, mut f: impl FnMut(&O::Value) -> bool) -> usize {
151        debug_assert!(f(&O::identity()));
152        let Self {
153            len,
154            offset,
155            ref values,
156        } = *self;
157        let mut end = start + offset;
158        let mut value = O::identity();
159        loop {
160            let p = end.trailing_zeros();
161            let end1 = end + (1 << p);
162            if offset + len < end1 {
163                break;
164            }
165            let value1 = O::op(&value, &values[end >> p]);
166            if !f(&value1) {
167                break;
168            }
169            end = end1;
170            value = value1;
171        }
172        for p in (0..end.trailing_zeros()).rev() {
173            let end1 = end + (1 << p);
174            if offset + len < end1 {
175                continue;
176            }
177            let value1 = O::op(&value, &values[end >> p]);
178            if !f(&value1) {
179                continue;
180            }
181            end = end1;
182            value = value1;
183        }
184        end - offset
185    }
186
187    /// Returns the minimum $l$ s.t. $f(x _ l \cdot \dots \cdot x _ { r - 1 })$ is `true`, if $f$ is monotone.
188    ///
189    /// More precisely, $r$ satisfies:
190    ///
191    /// 1. $f(x _ l \cdot \dots \cdot x _ { r - 1 })$ is `true`, or $l = r$
192    /// 2. $f(x _ { l - 1 } \cdot \dots \cdot x _ { r - 1 } )$ is `false`, or $l = 0$
193    pub fn min_left(&self, end: usize, mut f: impl FnMut(&O::Value) -> bool) -> usize {
194        debug_assert!(f(&O::identity()));
195        if end == 0 {
196            return 0;
197        }
198        let Self {
199            offset, ref values, ..
200        } = *self;
201        let mut start = end + offset;
202        let mut value = O::identity();
203        loop {
204            let p = (start | offset).trailing_zeros();
205            let start1 = start - (1 << p);
206            let value1 = O::op(&value, &values[start1 >> p]);
207            if !f(&value1) {
208                break;
209            }
210            start = start1;
211            value = value1;
212            if start == offset {
213                return 0;
214            }
215        }
216        for p in (0..(start | offset).trailing_zeros()).rev() {
217            let start1 = start - (1 << p);
218            let value1 = O::op(&value, &values[start1 >> p]);
219            if !f(&value1) {
220                continue;
221            }
222            start = start1;
223            value = value1;
224        }
225        start - offset
226    }
227
228    /// Returns the entry of $x_i$.
229    pub fn entry(&mut self, index: usize) -> Entry<'_, O> {
230        let offset = self.offset;
231        Entry {
232            segtree: self,
233            index: offset + index,
234        }
235    }
236
237    /// Returns an iterator of $x_0, x_1, \ldots, x_{n-1}$.
238    pub fn iter(&self) -> std::slice::Iter<'_, O::Value> {
239        self.into_iter()
240    }
241
242    /// Returns a slice of $x_0, x_1, \ldots, x_{n-1}$.
243    pub fn as_slice(&self) -> &[O::Value] {
244        &self.values[self.offset..self.offset + self.len]
245    }
246}

286/// The result of [`Segtree::entry`].
287pub struct Entry<'a, O: Op> {
288    segtree: &'a mut Segtree<O>,
289    index: usize,
290}
291impl<O: Op> Drop for Entry<'_, O> {
292    fn drop(&mut self) {
293        let mut index = self.index;
294        while index != 0 {
295            index /= 2;
296            self.segtree.values[index] = O::op(
297                &self.segtree.values[index * 2],
298                &self.segtree.values[index * 2 + 1],
299            );
300        }
301    }
302}
303impl<O: Op> Deref for Entry<'_, O> {
304    type Target = O::Value;
305
306    fn deref(&self) -> &Self::Target {
307        &self.segtree.values[self.index]
308    }
309}
310impl<O: Op> DerefMut for Entry<'_, O> {
311    fn deref_mut(&mut self) -> &mut Self::Target {
312        &mut self.segtree.values[self.index]
313    }
314}

318{
319    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {

324/// A sparse (compressed) segment tree.
325/// Use [`collect_map()`](Self::collect_map()) to debug.
326pub struct SegtreeWithCompression<K, O: Op> {
327    inner: Segtree<O>,
328    keys: Vec<K>,
329}
330impl<K: Ord, O: Op> SegtreeWithCompression<K, O> {
331    /// Constructs from the specified key-value pairs.
332    pub fn from_slice(kv: &[(K, O::Value)]) -> Self
333    where
334        K: Clone,
335        O::Value: Clone,
336    {
337        let mut kv = kv.to_vec();
338        kv.sort_by(|(a, _), (b, _)| a.cmp(b));
339        let (keys, values): (Vec<K>, Vec<O::Value>) = kv.into_iter().unzip();
340        Self {
341            inner: Segtree::from_slice(&values),
342            keys,
343        }
344    }
345
346    /// Folds $\left \lbrace x_k \mid k \in \text{{range}} \right \rbrace$.
347    pub fn fold<R: RangeBounds<K>>(&self, range: R) -> O::Value {
348        let (start, end) = to_range(range, &self.keys);
349        self.inner.fold(start..end)
350    }
351
352    /// Returns the entry of $x_k$.
353    /// If $k$ is not found, it panics.
354    pub fn entry(&mut self, key: &K) -> Entry<'_, O> {
355        let index = self.keys.binary_search(key).unwrap();
356        self.inner.entry(index)
357    }
358
359    /// Returns the keys.
360    pub fn keys(&self) -> &[K] {
361        &self.keys
362    }
363
364    /// Returns an iterator of $(k, x_k)$.
365    pub fn iter(&self) -> impl Iterator<Item = (&K, &O::Value)> {
366        self.keys.iter().zip(self.inner.as_slice())
367    }
368
369    /// Collects the key-value pairs into a `BTreeMap`.
370    pub fn collect_map(&self) -> BTreeMap<K, O::Value>
371    where
372        K: Clone,
373        O::Value: Clone,
374    {
375        self.keys
376            .iter()
377            .cloned()
378            .zip(self.inner.iter().cloned())
379            .collect()
380    }
381}
382
383impl<K, O: Op> fmt::Debug for SegtreeWithCompression<K, O>
384where
385    K: fmt::Debug,
386    O::Value: fmt::Debug,
387{
388    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
389        f.debug_struct("SparseSegtree")
390            .field("inner", &self.inner)
391            .field("keys", &self.keys)
392            .finish()
393    }
394}

414/// A segment tree of segment trees (2D segment tree).
415/// The multiplication must be commutative.
416pub struct Sparse2dSegtree<K, L, O: Op> {
417    segtrees: Vec<SegtreeWithCompression<L, O>>,
418    keys: Vec<K>,
419}
420impl<K, L, O: Op> Sparse2dSegtree<K, L, O>
421where
422    K: Ord + Clone,
423    L: Ord + Clone,
424    O::Value: Clone,
425{
426    /// Constructs from the specified key-value pairs.
427    pub fn from_slice(points: &[(K, L, O::Value)]) -> Self {
428        let mut keys = points.iter().map(|(k, _, _)| k.clone()).collect::<Vec<_>>();
429        keys.sort();
430        keys.dedup();
431        let mut lvs = vec![vec![]; keys.len() * 2];
432        for (k, l, v) in points {
433            let mut i = keys.binary_search(k).unwrap();
434            i += keys.len();
435            while i != 0 {
436                lvs[i].push((l.clone(), v.clone()));
437                i /= 2;
438            }
439        }
440        let segtrees = lvs
441            .into_iter()
442            .map(|lvs_| {
443                let mut ls = lvs_.iter().map(|(l, _)| l).collect::<Vec<_>>();
444                ls.sort();
445                ls.dedup();
446                let mut lvs = ls
447                    .iter()
448                    .map(|&l| (l.clone(), O::identity()))
449                    .collect::<Vec<_>>();
450                for (l, v) in &lvs_ {
451                    let i = ls.binary_search(&l).unwrap();
452                    lvs[i].1 = O::op(&lvs[i].1, v);
453                }
454                SegtreeWithCompression::from_slice(&lvs)
455            })
456            .collect::<Vec<_>>();
457        Self { segtrees, keys }
458    }
459
460    /// Folds $\left \lbrace x_{k, l} \mid (k, l) \in \text{{range}} \right \rbrace$.
461    pub fn fold(&self, i: impl RangeBounds<K>, j: impl RangeBounds<L> + Clone) -> O::Value {
462        let (mut i0, mut i1) = to_range(i, &self.keys);
463        i0 += self.keys.len();
464        i1 += self.keys.len();
465        let mut left = O::identity();
466        let mut right = O::identity();
467        while i0 < i1 {
468            if i0 % 2 == 1 {
469                left = O::op(&left, &self.segtrees[i0].fold(j.clone()));
470                i0 += 1;
471            }
472            if i1 % 2 == 1 {
473                i1 -= 1;
474                right = O::op(&self.segtrees[i1].fold(j.clone()), &right);
475            }
476            i0 /= 2;
477            i1 /= 2;
478        }
479        O::op(&left, &right)
480    }
481
482    /// Apply a function to $x_{k, l}$.
483    ///
484    /// The function $f$ must satisfy $f(x \cdot y) = x \cdot f(y)$.
485    pub fn apply(&mut self, k: &K, l: &L, mut f: impl FnMut(&mut O::Value)) {
486        let mut i = self.keys.binary_search(k).unwrap();
487        i += self.keys.len();
488        while i != 0 {
489            f(&mut self.segtrees[i].entry(l));
490            i /= 2;
491        }
492    }
493
494    /// Returns the iterator of $(k, l, x_{k, l})$.
495    pub fn iter(&self) -> impl Iterator<Item = (&K, &L, &O::Value)> {
496        self.keys
497            .iter()
498            .zip(self.segtrees[self.keys.len()..].iter())
499            .flat_map(|(k, segtree)| segtree.iter().map(move |(l, v)| (k, l, v)))
500    }
501
502    /// Collects the key-value pairs into a `BTreeMap`.
503    pub fn collect_map(&self) -> BTreeMap<(K, L), O::Value>
504    where
505        K: Clone,
506        L: Clone,
507        O::Value: Clone,
508    {
509        self.keys
510            .iter()
511            .flat_map(|k| {
512                self.segtrees[self.keys.len() + self.keys.binary_search(k).unwrap()]
513                    .iter()
514                    .map(move |(l, v)| ((k.clone(), l.clone()), v.clone()))
515            })
516            .collect()
517    }
518}

561/// A dense 2D segment tree.
562pub struct Dense2dSegtree<O: Op> {
563    values: Vec<Vec<O::Value>>,
564}
565impl<O: Op> Dense2dSegtree<O> {
566    /// Constructs from the specified values.
567    pub fn from_slice(values: &[Vec<O::Value>]) -> Self
568    where
569        O::Value: Clone,
570    {
571        let values_ = values;
572        let h = values.len();
573        let w = values.first().map_or(0, Vec::len);
574        let mut values = vec![vec![O::identity(); 2 * w]; 2 * h];
575        for (values, values_) in values[h..].iter_mut().zip(values_) {
576            values[w..].clone_from_slice(values_);
577            for j in (1..w).rev() {
578                values[j] = O::op(&values[j * 2], &values[j * 2 + 1]);
579            }
580        }
581        for i in (1..h).rev() {
582            for j in 0..2 * w {
583                values[i][j] = O::op(&values[i * 2][j], &values[i * 2 + 1][j]);
584            }
585        }
586        Self { values }
587    }
588
589    /// Fold $\left \lbrace x_{i, j} \mid i \in \text{{range}}_i, j \in \text{{range}}_j \right \rbrace$.
590    pub fn fold(&self, i: impl RangeBounds<usize>, j: impl RangeBounds<usize>) -> O::Value {
591        let h = self.values.len() / 2;
592        let w = self.values.first().map_or(0, |v| v.len() / 2);
593        let (mut i0, mut i1) = open(i, h);
594        assert!(i0 <= i1 && i1 <= h);
595        let (mut j0, mut j1) = open(j, w);
596        assert!(j0 <= j1 && j1 <= w);
597        i0 += h;
598        i1 += h;
599        j0 += w;
600        j1 += w;
601        let mut left = O::identity();
602        let mut right = O::identity();
603        while i0 < i1 {
604            if i0 % 2 == 1 {
605                let mut j0 = j0;
606                let mut j1 = j1;
607                while j0 < j1 {
608                    if j0 % 2 == 1 {
609                        left = O::op(&left, &self.values[i0][j0]);
610                        j0 += 1;
611                    }
612                    if j1 % 2 == 1 {
613                        j1 -= 1;
614                        right = O::op(&self.values[i0][j1], &right);
615                    }
616                    j0 /= 2;
617                    j1 /= 2;
618                }
619                i0 += 1;
620            }
621            if i1 % 2 == 1 {
622                i1 -= 1;
623                let mut j0 = j0;
624                let mut j1 = j1;
625                while j0 < j1 {
626                    if j0 % 2 == 1 {
627                        left = O::op(&left, &self.values[i1][j0]);
628                        j0 += 1;
629                    }
630                    if j1 % 2 == 1 {
631                        j1 -= 1;
632                        right = O::op(&self.values[i1][j1], &right);
633                    }
634                    j0 /= 2;
635                    j1 /= 2;
636                }
637            }
638            i0 /= 2;
639            i1 /= 2;
640        }
641        O::op(&left, &right)
642    }
643
644    /// Returns the entry of $x_{i, j}$.
645    pub fn entry(&mut self, i: usize, j: usize) -> Dense2dEntry<'_, O> {
646        let h = self.values.len() / 2;
647        let w = self.values.first().map_or(0, |v| v.len() / 2);
648        Dense2dEntry {
649            segtree: self,
650            i: h + i,
651            j: w + j,
652        }
653    }
654
655    /// Returns an iterator that returns the rows $(x_{i, 0}, x_{i, 1}, \ldots, x_{i, w-1})$.
656    pub fn iter(&self) -> impl Iterator<Item = &[O::Value]> {
657        self.values[self.values.len() / 2..]
658            .iter()
659            .map(|v| &v[v.len() / 2..])
660    }
661
662    /// Collect to a $2$-dimensional vector.
663    pub fn collect_vec(&self) -> Vec<Vec<O::Value>>
664    where
665        O::Value: Clone,
666    {
667        self.iter().map(<[_]>::to_vec).collect()
668    }

690/// The result of [`Dense2dSegtree::entry`].
691pub struct Dense2dEntry<'a, O: Op> {
692    segtree: &'a mut Dense2dSegtree<O>,
693    i: usize,
694    j: usize,
695}
696impl<O: Op> Drop for Dense2dEntry<'_, O> {
697    fn drop(&mut self) {
698        let mut i = self.i;
699        let mut j = self.j / 2;
700        while j != 0 {
701            self.segtree.values[i][j] = O::op(
702                &self.segtree.values[i][2 * j],
703                &self.segtree.values[i][2 * j + 1],
704            );
705            j /= 2;
706        }
707        i /= 2;
708        while i != 0 {
709            let mut j = self.j;
710            while j != 0 {
711                self.segtree.values[i][j] = O::op(
712                    &self.segtree.values[i * 2][j],
713                    &self.segtree.values[i * 2 + 1][j],
714                );
715                j /= 2;
716            }
717            i /= 2;
718        }
719    }
720}
721impl<O: Op> Deref for Dense2dEntry<'_, O> {
722    type Target = O::Value;
723
724    fn deref(&self) -> &Self::Target {
725        &self.segtree.values[self.i][self.j]
726    }
727}
728impl<O: Op> DerefMut for Dense2dEntry<'_, O> {
729    fn deref_mut(&mut self) -> &mut Self::Target {
730        &mut self.segtree.values[self.i][self.j]
731    }
732}

734fn open<B: RangeBounds<usize>>(bounds: B, n: usize) -> (usize, usize) {
735    use std::ops::Bound;
736    let start = match bounds.start_bound() {
737        Bound::Unbounded => 0,
738        Bound::Included(&x) => x,
739        Bound::Excluded(&x) => x + 1,
740    };
741    let end = match bounds.end_bound() {
742        Bound::Unbounded => n,
743        Bound::Included(&x) => x + 1,
744        Bound::Excluded(&x) => x,
745    };
746    (start, end)
747}
748
749fn to_range<K: Ord, B: RangeBounds<K>>(bounds: B, keys: &[K]) -> (usize, usize) {
750    use std::ops::Bound;
751    let start = match bounds.start_bound() {
752        Bound::Unbounded => 0,
753        Bound::Included(x) => match keys.binary_search(x) {
754            Ok(i) | Err(i) => i,
755        },
756        Bound::Excluded(x) => match keys.binary_search(x) {
757            Ok(i) => i + 1,
758            Err(i) => i,
759        },
760    };
761    let end = match bounds.end_bound() {
762        Bound::Unbounded => keys.len(),
763        Bound::Included(x) => match keys.binary_search(x) {
764            Ok(i) => i + 1,
765            Err(i) => i,
766        },
767        Bound::Excluded(x) => match keys.binary_search(x) {
768            Ok(i) | Err(i) => i,
769        },
770    };
771    (start, end)
772}
